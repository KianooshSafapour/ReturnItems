    75→    75→/** برچسب کوتاه شمسی محور افقی: «۱۲ مهر» */
    76→    76→function shortJalali(isoDate: string): string {
    77→    77→  const [y, m, d] = isoDate.split("-").map((p) => parseInt(p, 10));
    78→    78→  const j = toJalali(new Date(y || 2025, (m || 1) - 1, d || 1));
    79→    79→  return `${toFaDigits(j.jd)} ${JALALI_MONTHS[j.jm - 1]}`;
    80→    80→}

    82→    82→/** کوتاه‌سازی نام کالا برای برچسب محور عمودی */
    83→    83→function truncateName(name: string, max = 24): string {
    84→    84→  return name.length > max ? `${name.slice(0, max - 1)}…` : name;
    85→    85→}
