   169→      if (apiKey.trim()) payload.apiKey = apiKey.trim();
   170→      if (username.trim() || config?.username) payload.username = username.trim();
   171→      if (password.trim()) payload.password = password.trim();

   173→        payload.customUrl = customUrl;
   174→        payload.customMethod = customMethod;
   175→        payload.customHeaders = customHeaders;
   176→        payload.customBody = customBody;
   177→      }

   179→        method: "PUT",
   180→        headers: { "Content-Type": "application/json" },
   181→        body: JSON.stringify(payload),
   182→      });

   191→      }
   192→    } catch {
   193→      toast.error("ارتباط با سرور برقرار نشد");
   194→    } finally {
   195→      setSaving(false);

   969→function SmsTemplateDialog({
   970→  open,
   971→  target,
   972→  onClose,
   973→  onSaved,
   974→}: {
   975→  open: boolean;
   976→  target: SmsTemplateDTO | null;
   977→  onClose: () => void;
   978→  onSaved: () => void;
   979→}) {
   980→  const [title, setTitle] = useState("");
   981→  const [body, setBody] = useState("");
   982→  const [isActive, setIsActive] = useState(true);
   983→  const [saving, setSaving] = useState(false);
   984→  const [error, setError] = useState<string | null>(null);
   985→
   986→  useEffect(() => {
   987→    if (open) {
   988→      setTitle(target?.title ?? "");
   989→      setBody(target?.body ?? "");
   990→      setIsActive(target?.isActive ?? true);
   991→      setError(null);
   992→    }
   993→  }, [open, target]);
   994→
   995→  const seg = smsSegments(body);
   996→
   997→  async function save() {
   998→    if (saving) return;
   999→    const t = title.trim();
  1000→    const b = body.trim();
  1001→    if (t.length < 2) {
  1002→      setError("نام نوع پیام حداقل ۲ نویسه باشد");
  1003→      return;
  1004→    }
  1005→    if ([...b].length < 5) {
  1006→      setError("متن پیام حداقل ۵ نویسه باشد");
  1007→      return;
  1008→    }
  1009→    setSaving(true);
  1010→    try {
  1011→      const res = await fetch(
  1012→        target ? `/api/sms/templates/${target.id}` : "/api/sms/templates",
  1013→        {
  1014→          method: target ? "PATCH" : "POST",

  1017→        }
  1018→      );
  1019→      const d = await res.json();
  1020→      if (res.ok) {
  1021→        toast.success(target ? "قالب پیام ویرایش شد" : "نوع پیام جدید تعریف شد");
  1022→        onSaved();
  1023→      } else {
  1024→        setError(d.error || "خطا در ذخیره قالب");
  1025→      }
  1026→    } catch {
  1027→      setError("ارتباط با سرور برقرار نشد");
  1028→    } finally {
  1029→      setSaving(false);
  1030→    }
  1031→  }
  1032→
  1033→  return (
  1034→    <Dialog open={open} onOpenChange={(o) => !saving && !o && onClose()}>

  1039→            {target ? `ویرایش قالب «${target.title}»` : "تعریف نوع پیام جدید"}
  1040→          </DialogTitle>
  1041→          <DialogDescription>
  1042→            {target
  1043→              ? "متن و نام نوع پیام را ویرایش کنید."
  1044→              : "نوع پیام جدیدی برای اطلاع‌رسانی به مشتریان تعریف کنید (مثلاً «کالای معیوب» یا «بازپرداخت اقساط»)."}
  1045→          </DialogDescription>
  1046→        </DialogHeader>
  1047→
  1048→        <div className="space-y-4">
  1049→          <div className="space-y-2">
  1050→            <Label htmlFor="tpl-title">نام نوع پیام</Label>
  1051→            <Input
  1052→              id="tpl-title"
  1053→              className="h-11"
  1054→              placeholder="مثلاً: مرجوعی کالا"
  1055→              value={title}
  1056→              maxLength={60}
  1057→              onChange={(e) => setTitle(e.target.value)}
  1058→            />
  1059→          </div>
  1060→
  1061→          <div className="space-y-2">
  1062→            <div className="flex items-center justify-between gap-2">
  1063→              <Label htmlFor="tpl-body">متن پیام</Label>
  1064→              {body && (
  1065→                <span className="num text-[10px] text-muted-foreground">
  1066→                  {toFaDigits(seg.chars)} نویسه — {toFaDigits(seg.parts)} پیامک
  1067→                </span>
  1068→              )}
  1069→            </div>
  1070→            <Textarea
  1071→              id="tpl-body"
  1072→              dir="rtl"
  1073→              rows={5}

  1113→        <DialogFooter className="gap-2">
