    18→    18→/* ---------- بارکد SVG (EAN-13 استاندارد یا CODE128 برای سایر مقادیر) ---------- */
    19→    19→function BarcodeSvg({
    20→    20→  value,
    21→    21→  height = 30,
    22→    22→  width = 1.6,
    23→    23→  fontSize = 12,
    24→    24→  className = "",
    25→    25→}: {
    26→    26→  value: string;
    27→    27→  height?: number;
    28→    28→  width?: number;
    29→    29→  fontSize?: number;
    30→    30→  className?: string;
    31→    31→}) {
    32→    32→  const ref = useRef<SVGSVGElement>(null);
    33→    33→
    34→    34→  useEffect(() => {
    35→    35→    if (!ref.current || !value) return;
    36→    36→    const isEan13 = /^\d{13}$/.test(value) && isValidEan13(value);
    37→    37→    try {
    38→    38→      JsBarcode(ref.current, value, {
    39→    39→        format: isEan13 ? "EAN13" : "CODE128",
    40→    40→        height,
    41→    41→        width,
    42→    42→        fontSize,
    43→    43→        margin: 2,
    44→    44→        lineColor: "#111111",
    45→    45→        background: "transparent",
    46→    46→        displayValue: true,
    47→    47→        font: "monospace",
    48→    48→        fontOptions: "bold",
    49→    49→        textAlign: "center",
    50→    50→        textPosition: "bottom",
    51→    51→        textMargin: 1,
    52→    52→      });
    53→    53→    } catch {
    54→    54→      // مقدار نامعتبر برای بارکد → نمایش متن خام
    55→    55→      if (ref.current) ref.current.innerHTML = "";
    56→    56→    }
    57→    57→  }, [value, height, width, fontSize]);
    58→    58→
    59→    59→  return <svg ref={ref} className={className} aria-label={`بارکد ${value}`} />;
    60→    60→}

   280→   280→export function ReceiptDialog({
   281→   281→  open,
   282→   282→  onOpenChange,
   283→   283→  detail,
   284→   284→}: {
   285→   285→  open: boolean;
   286→   286→  onOpenChange: (o: boolean) => void;
   287→   287→  detail: ReturnDetailDTO | null;
   288→   288→}) {
   289→   289→  const [settings, setSettings] = useState<StoreSettings>(DEFAULT_SETTINGS);
   290→   290→  // قالب چاپ: برگه A4 یا نوار حرارتی ۸۰ میلی‌متری — انتخاب آخرین بار ذخیره می‌شود
   291→   291→  const [format, setFormat] = useState<"a4" | "thermal">(() => {
   292→   292→    if (typeof window === "undefined") return "a4";
   293→   293→    try {
   294→   294→      return window.localStorage.getItem("hz:receipt-format") === "thermal" ? "thermal" : "a4";
   295→   295→    } catch {
   296→   296→      return "a4";
   297→   297→    }
   298→   298→  });
   299→   299→
   300→   300→  function chooseFormat(f: "a4" | "thermal") {
   301→   301→    setFormat(f);
   302→   302→    try {
   303→   303→      window.localStorage.setItem("hz:receipt-format", f);
   304→   304→    } catch {
   305→   305→      /* حافظه در دسترس نیست */
   306→   306→    }
   307→   307→  }
   308→   308→
   309→   309→  // تنظیمات فروشگاه برای سربرگ/پابرگ رسید
   310→   310→  useEffect(() => {
   311→   311→    if (!open) return;
   312→   312→    let cancelled = false;
   313→   313→    fetch("/api/admin/settings")
   314→   314→      .then((r) => (r.ok ? r.json() : null))
   315→   315→      .then((d) => {
   316→   316→        if (!cancelled && d && typeof d === "object") setSettings({ ...DEFAULT_SETTINGS, ...d });
   317→   317→      })
   318→   318→      .catch(() => {});
   319→   319→    return () => {
   320→   320→      cancelled = true;
   321→   321→    };
   322→   322→  }, [open]);
   323→   323→
   324→   324→  const paid = detail?.status === "paid" || detail?.status === "approved";
   325→   325→  const settlement =
   326→   326→    detail && (detail.transferAmount !== detail.totalAmount || detail.exchangeValue > 0)
   327→   327→      ? {
   328→   328→          T: detail.totalAmount,
   329→   329→          E: detail.exchangeValue,
   330→   330→          X: detail.transferAmount,
   331→   331→          debt: detail.exchangeValue > detail.totalAmount ? detail.exchangeValue - detail.totalAmount : 0,
   332→   332→          credit:
   333→   333→            detail.transferAmount + detail.exchangeValue < detail.totalAmount
   334→   334→              ? detail.totalAmount - detail.exchangeValue - detail.transferAmount
   335→   335→              : 0,
   336→   336→        }
   337→   337→      : null;
   338→   338→
   339→   339→  // ---- اندازهٔ صفحهٔ چاپ حرارتی: به‌اندازهٔ محتوای واقعی (نه ارتفاع ثابت) ----
   340→   340→  // قبلاً @page 80mm×۳۲۷mm بود → کاغذ حرارتی صفحهٔ ثابت ۳۲۷ میلی‌متری چاپ می‌کرد و
   341→   341→  // تا پایان رول کاغذ سفید هدر می‌داد. حالا ارتفاع از محتوای رندرشده اندازه‌گیری می‌شود:
   342→   342→  // چاپ از بالای رسید شروع و دقیقاً جایی که محتوا تمام می‌شود قطع می‌شود.
   343→   343→  const thermalStyleRef = useRef<HTMLStyleElement>(null);
   344→   344→  const thermalPaperRef = useRef<HTMLDivElement>(null);
   345→   345→
   346→   346→  function syncThermalPageSize() {
   347→   347→    if (format !== "thermal") return;
   348→   348→    const paper = thermalPaperRef.current;
   349→   349→    const styleEl = thermalStyleRef.current;
   350→   350→    if (!paper || !styleEl) return;
   351→   351→    // تبدیل پیکسل CSS به میلی‌متر (۹۶ پیکسل = ۱ اینچ = ۲۵٫۴ میلی‌متر)
   352→   352→    const px = Math.ceil(paper.getBoundingClientRect().height);
   353→   353→    const mm = Math.max(30, Math.round((px * 25.4) / 96 + 3)); // حداقل ۳۰mm + بافر ۳mm برای گرد شدن
   354→   354→    styleEl.textContent = `@media print { @page { size: 80mm ${mm}mm; margin: 0; } }`;
   355→   355→  }
   356→   356→
   357→   357→  // همگام‌سازی پس از رندر (باز شدن دیالوگ / تغییر قالب / بارکدها)
   358→   358→  useEffect(() => {
   359→   359→    if (!open || format !== "thermal") return;
   360→   360→    const t1 = requestAnimationFrame(() => requestAnimationFrame(syncThermalPageSize));
   361→   361→    const t2 = window.setTimeout(syncThermalPageSize, 350); // پس از رندر نهایی بارکدها
   362→   362→    return () => {
   363→   363→      cancelAnimationFrame(t1);
   364→   364→      window.clearTimeout(t2);
   365→   365→    };
   366→   366→  }, [open, format, detail?.id]);
   367→   367→
   368→   368→  function handlePrint() {
