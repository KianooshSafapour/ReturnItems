   351→   351→function AddUserDialog({
   352→   352→  open,
   353→   353→  onOpenChange,
   354→   354→  onCreated,
   355→   355→}: {
   356→   356→  open: boolean;
   357→   357→  onOpenChange: (o: boolean) => void;
   358→   358→  onCreated: () => void;
   359→   359→}) {
   360→   360→  const [fullName, setFullName] = useState("");
   361→   361→  const [username, setUsername] = useState("");
   362→   362→  const [phone, setPhone] = useState("");
   363→   363→  const [password, setPassword] = useState("");
   364→   364→  const [role, setRole] = useState<Role>("cashier");
   365→   365→  const [showPass, setShowPass] = useState(false);
   366→   366→  const [errors, setErrors] = useState<FormErrors>({});
   367→   367→  const [submitting, setSubmitting] = useState(false);
   368→   368→
   369→   369→  useEffect(() => {
   370→   370→    if (open) {
   371→   371→      setFullName("");
   372→   372→      setUsername("");
   373→   373→      setPhone("");
   374→   374→      setPassword("");
   375→   375→      setRole("cashier");
   376→   376→      setShowPass(false);
   377→   377→      setErrors({});
   378→   378→    }
   379→   379→  }, [open]);
   380→   380→
   381→   381→  const validate = (): boolean => {
   382→   382→    const e: FormErrors = {};
   383→   383→    if (fullName.trim().length < 3) e.fullName = "نام کامل حداقل ۳ نویسه باشد";
   384→   384→    if (!/^[a-z0-9_.-]{3,30}$/.test(username.trim().toLowerCase()))
   385→   385→      e.username = "نام کاربری باید ۳ تا ۳۰ نویسه لاتین کوچک، عدد، نقطه، خط تیره یا زیرخط باشد";
   386→   386→    const ph = digitsOnly(toEnDigits(phone));
   387→   387→    if (ph && !/^09\d{9}$/.test(ph)) e.phone = "شماره موبایل باید ۱۱ رقم و با ۰۹ شروع شود";
   388→   388→    if (password.length < 6) e.password = "گذرواژه حداقل ۶ نویسه باشد";
   389→   389→    setErrors(e);
   390→   390→    return Object.keys(e).length === 0;
   391→   391→  };
   392→   392→
   393→   393→  const submit = async () => {
   394→   394→    if (!validate() || submitting) return;
   395→   395→    setSubmitting(true);
   396→   396→    try {
   397→   397→      const res = await fetch("/api/users", {
   398→   398→        method: "POST",
   399→   399→        headers: { "Content-Type": "application/json" },
   400→   400→        body: JSON.stringify({
   401→   401→          fullName: fullName.trim(),
   402→   402→          username: username.trim().toLowerCase(),
   403→   403→          phone: digitsOnly(toEnDigits(phone)).trim(),
   404→   404→          password,
   405→   405→          role,
   406→   406→        }),
   407→   407→      });
   408→   408→      const data = await res.json();
   409→   409→      if (res.ok) {
