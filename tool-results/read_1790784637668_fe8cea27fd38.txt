   278→// ---------- پنجره جزئیات مرجوعی ----------
   279→export function ReturnDetailDialog({
   280→  id,
   281→  onClose,
   282→  user,
   283→  onPaid,
   284→}: {
   285→  id: string | null;
   286→  onClose: () => void;
   287→  user: SessionUser;
   288→  onPaid?: () => void;
   289→}) {
   290→  // الگوی استخراج حالت: بارگذاری از مقایسهٔ شناسهٔ درخواستی با دادهٔ واکشی‌شده به‌دست می‌آید
   291→  const [fetched, setFetched] = useState<{ id: string; data: ReturnDetailDTO | null } | null>(null);
   292→  const current = fetched && fetched.id === id ? fetched : null;
   293→  const loading = !!id && !current;
   294→  const detail = current?.data ?? null;
   295→
   296→  useEffect(() => {
   297→    if (!id) return;
   298→    let cancelled = false;
   299→    fetch(`/api/returns/${id}`)
   300→      .then((r) => r.json())
   301→      .then((d) => {
   302→        if (cancelled) return;
   303→        if (d.id) setFetched({ id, data: d });
   304→        else {
   305→          setFetched({ id, data: null });
   306→          toast.error(d.error || "خطا در دریافت جزئیات");
   307→        }
   308→      })
   309→      .catch(() => {
   310→        if (!cancelled) {
   311→          setFetched({ id, data: null });
   312→          toast.error("خطا در دریافت جزئیات");
   313→        }
   314→      });
   315→    return () => {
   316→      cancelled = true;
   317→    };
   318→  }, [id]);

   320→  // پاک‌سازی جزئیات هنگام بستن پنجره (در رویداد، نه افکت)

   322→    setFetched(null);

   328→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto p-0" dir="rtl">
   329→        {loading || !detail ? (
   330→          <div className="p-10 flex flex-col items-center gap-4">
   331→            <DialogHeader className="sr-only">
   332→              <DialogTitle>جزئیات مرجوعی</DialogTitle>

   334→            </DialogHeader>
   335→            <Loader2 className="size-8 animate-spin text-primary" />
   336→            <p className="text-sm text-muted-foreground">در حال دریافت جزئیات…</p>
   337→          </div>
   338→        ) : (
   339→          <div>
   340→            <DialogHeader className="px-6 pt-6 pb-0">
   341→              <DialogTitle className="flex items-center gap-2 flex-wrap">
   342→                <span className="num text-primary" dir="ltr">
   343→                  {detail.code}
   344→                </span>
   345→                {detail.status === "pending" ? (
   346→                  <Badge className="bg-chart-2/15 text-chart-2 border-chart-2/30 gap-1">
   347→                    <Hourglass className="size-3" /> در انتظار پرداخت
   348→                  </Badge>
   349→                ) : (
   350→                  <Badge className="bg-primary/15 text-primary border-primary/30 gap-1">
   351→                    <CheckCircle2 className="size-3" /> پرداخت و بایگانی شد
   352→                  </Badge>
   353→                )}
   354→              </DialogTitle>
   355→              <DialogDescription className="sr-only">
   356→                جزئیات کامل مرجوعی، اقلام و اطلاعات پرداخت

   358→            </DialogHeader>
   359→
   360→            <div className="p-6 space-y-6">
   361→              {/* اطلاعات مشتری */}
   362→              <section className="grid sm:grid-cols-2 gap-3 text-sm">

   364→                <InfoRow
   365→                  icon={<Phone className="size-4" />}
   366→                  label="موبایل"
   367→                  value={toFaDigits(detail.customerPhone)}

   705→export function PaymentForm({
   706→  detail,
   707→  onPaid,
   708→}: {
   709→  detail: ReturnDetailDTO;
   710→  onPaid: (code: string, noTransfer?: boolean) => void;
   711→}) {
   712→  // حالت تسویه بدون انتقال وجه — تبادل کالا / بدهی مشتری / اعتبار
   713→  const noTransfer = detail.transferAmount === 0;
   714→  const [bankName, setBankName] = useState("");
   715→  const [accountNumber, setAccountNumber] = useState("");
   716→  const [cardNumber, setCardNumber] = useState("");
   717→  const [referenceNumber, setReferenceNumber] = useState("");
   718→  const [transferredAt, setTransferredAt] = useState<Date | null>(null);
   719→  const [submitting, setSubmitting] = useState(false);
   720→
   721→  async function submit() {
   722→    if (submitting) return;
   723→    if (!transferredAt) {
   724→      toast.error(noTransfer ? "زمان تسویه را انتخاب کنید" : "زمان انتقال وجه را انتخاب کنید");
   725→      return;
   726→    }
   727→    if (!noTransfer) {
   728→      if (!bankName) {
   729→        toast.error("بانک مبدأ را انتخاب کنید");
   730→        return;
   731→      }
   732→      if (!accountNumber && !digitsOnly(cardNumber) && !referenceNumber) {
   733→        toast.error("حداقل شماره حساب، کارت یا شماره پیگیری را وارد کنید");
   734→        return;
   735→      }
   736→      if (cardNumber && !/^\d{16}$/.test(digitsOnly(cardNumber))) {
   737→        toast.error("شماره کارت مبدأ باید ۱۶ رقم باشد");
   738→        return;
   739→      }
   740→    }
   741→    setSubmitting(true);
   742→    try {
   743→      const res = await fetch(`/api/returns/${detail.id}/payment`, {
   744→        method: "POST",
   745→        headers: { "Content-Type": "application/json" },
   746→        body: JSON.stringify(
   747→          noTransfer
   748→            ? { noTransfer: true, transferredAt: transferredAt.toISOString() }
   749→            : {
   750→                bankName,
   751→                accountNumber: accountNumber.trim() || null,
   752→                cardNumber: digitsOnly(cardNumber) || null,
   753→                referenceNumber: referenceNumber.trim() || null,
   754→                transferredAt: transferredAt.toISOString(),
   755→              }
   756→        ),
   757→      });
   758→      const data = await res.json();
   759→      if (!res.ok) {
   760→        toast.error(data.error || "ثبت پرداخت ناموفق بود");
   761→        return;
   762→      }
   763→      onPaid(data.code, noTransfer);
   764→    } catch {
   765→      toast.error("ارتباط با سرور برقرار نشد");
   766→    } finally {
   767→      setSubmitting(false);
   768→    }
   769→  }
   770→
   771→  const amountDiffers = detail.transferAmount !== detail.totalAmount || detail.exchangeValue > 0;
   772→
   773→  return (
   774→    <section className="rounded-xl border-2 border-primary/30 bg-card p-4 space-y-4">
   775→      <div className="flex items-center gap-2 flex-wrap">
   776→        <Wallet className="size-4 text-primary" />
   777→        <h4 className="font-bold text-sm">ثبت بازگشت وجه (ویژه مدیریت)</h4>
   778→        <SettlementBadge type={detail.settlementType} className="mr-auto" />
   779→      </div>
   780→      <IlluminatedDivider />
   781→      <div className="rounded-lg bg-primary/5 border border-primary/20 p-3 space-y-2">
   782→        <div className="flex items-center gap-1.5">
   783→          <Wallet className="size-3.5 text-primary" />
   784→          <p className="text-xs font-bold text-primary">مبلغ قابل انتقال به حساب مشتری</p>
   785→        </div>
   786→        <MoneyDisplay amount={detail.transferAmount} size="md" showWords={detail.transferAmount > 0} />
   787→        {amountDiffers && (
   788→          <div className="pt-2 border-t border-primary/15 space-y-1 text-[11px]">
   789→            <p className="flex items-center justify-between gap-2 flex-wrap">
   790→              <span className="text-muted-foreground">جمع اقلام مرجوعی</span>
   791→              <span className="num font-semibold">{formatMoney(detail.totalAmount)} ریال</span>
   792→            </p>
   793→            {detail.exchangeValue > 0 && (
   794→              <p className="flex items-center justify-between gap-2 flex-wrap">
   795→                <span className="text-muted-foreground flex items-center gap-1">
   796→                  <Repeat2 className="size-3 text-chart-3" />
   797→                  ارزش کالای جایگزین
   798→                </span>
   799→                <span className="num font-semibold text-chart-3">− {formatMoney(detail.exchangeValue)} ریال</span>
   800→              </p>
   801→            )}
   802→          </div>
   803→        )}
   804→      </div>
   805→      {noTransfer ? (
   806→        <>
   807→          {/* حالت تسویه بدون انتقال وجه — فقط زمان تسویه */}
   808→          <div className="rounded-xl border border-chart-3/40 bg-chart-3/10 p-3.5 flex items-start gap-2.5" role="status">
   809→            <Info className="size-4 text-chart-3 shrink-0 mt-0.5" />
   810→            <p className="text-xs leading-6">
   811→              این مرجوعی بدون انتقال وجه تسویه می‌شود (تبادل کالا / بدهی مشتری / اعتبار).
   812→              برای بایگانی، زمان تسویه را انتخاب کنید.
   813→            </p>
   814→          </div>
   815→          <div className="space-y-1.5">
   816→            <label className="text-xs font-semibold">زمان تسویه</label>
   817→            <JalaliDateTimePicker value={transferredAt} onChange={setTransferredAt} placeholder="انتخاب زمان تسویه" />
   818→          </div>
   819→        </>
   820→      ) : (
   821→        <div className="grid sm:grid-cols-2 gap-3">
   822→          <div className="space-y-1.5">
   823→            <label className="text-xs font-semibold flex items-center gap-1">
   824→              <Landmark className="size-3.5 text-primary" />
   825→              بانک مبدأ (انتقال وجه)
   826→            </label>
   827→            <Select value={bankName} onValueChange={setBankName}>
   828→              <SelectTrigger className="h-10">
   829→                <SelectValue placeholder="انتخاب بانک…" />
   830→              </SelectTrigger>
   831→              <SelectContent className="max-h-72">
   832→                {BANKS.map((b) => (
   833→                  <SelectItem key={b} value={b}>
   834→                    {b}
   835→                  </SelectItem>
   836→                ))}
   837→              </SelectContent>
   838→            </Select>
   839→          </div>
   840→          <div className="space-y-1.5">
   841→            <label className="text-xs font-semibold">زمان انتقال وجه</label>
   842→            <JalaliDateTimePicker value={transferredAt} onChange={setTransferredAt} placeholder="انتخاب زمان انتقال" />
   843→          </div>
   844→          <div className="space-y-1.5">
   845→            <label className="text-xs font-semibold">شماره حساب مبدأ</label>

   847→              dir="ltr"
   848→              className="num text-left h-10"
   849→              placeholder="اختیاری — مثال: 0101234567890"
   850→              value={accountNumber ? toFaDigits(accountNumber) : ""}
   851→              onChange={(e) => setAccountNumber(toEnDigits(e.target.value).replace(/[^\d]/g, "").slice(0, 26))}
   852→            />
   853→          </div>
   854→          <div className="space-y-1.5">
   855→            <label className="text-xs font-semibold">شماره کارت مبدأ</label>
   856→            <Input
   857→              dir="ltr"
   858→              inputMode="numeric"
   859→              className="num text-left h-10"
   860→              placeholder="اختیاری — ۱۶ رقم"
   861→              value={cardNumber ? formatCardNumber(cardNumber) : ""}
   862→              onChange={(e) => setCardNumber(digitsOnly(e.target.value).slice(0, 16))}
   863→            />
   864→          </div>
   865→          <div className="space-y-1.5 sm:col-span-2">
   866→            <label className="text-xs font-semibold flex items-center gap-1">
   867→              <Hash className="size-3.5 text-primary" />
   868→              شماره پیگیری/ارجاع انتقال
   869→            </label>
   870→            <Input
   871→              dir="ltr"
   872→              className="text-left h-10"
   873→              placeholder="اختیاری — شماره ارجاع بانک"
   874→              value={referenceNumber}
   875→              onChange={(e) => setReferenceNumber(e.target.value.slice(0, 40))}
   876→            />
   877→          </div>
   878→        </div>
   879→      )}
   880→      <Button className="w-full h-12 font-bold" onClick={submit} disabled={submitting}>

   885→        ) : (

   887→            <CheckCircle2 className="size-4" />
   888→            {noTransfer ? "تسویه و بایگانی بدون انتقال وجه" : "ثبت پرداخت و بایگانی مرجوعی"}
   889→          </>
   890→        )}
   891→      </Button>
   892→    </section>
   893→  );
