   166→  function addItem(p: ProductDTO | null, manual = false): boolean {
   167→    let existing: ReturnItemDraft | undefined;
   168→    if (p) {
   169→      existing = items.find(
   170→        (i) => (p.barcode && i.barcode === p.barcode) || (p.id && i.productId === p.id)
   171→      );

   173→        setItems((prev) =>
   174→          prev.map((i) => (i.key === existing!.key ? { ...i, quantity: Math.min(999, i.quantity + 1) } : i))
   175→        );

   177→          description: "تعداد فعلی: " + toFaDigits((existing.quantity || 0) + 1),
   178→        });
   179→        return true;
   180→      }
   181→      setItems((prev) => [
   182→        {
   183→          key: `k${keySeq++}`,
   184→          productId: p.id,
   185→          name: p.name,
   186→          barcode: p.barcode,
   187→          code: p.code,
   188→          quantity: 1,
   189→          unitPrice: p.price,
   190→          fromDb: true,
   191→          barcodeLocked: !!p.barcode,
   192→        },
   193→        ...prev,
   194→      ]);
   195→      return true;
   196→    }
   197→    if (manual) {
   198→      setItems((prev) => [
   199→        {
   200→          key: `k${keySeq++}`,
   201→          productId: null,
   202→          name: "",

   210→        },
   211→        ...prev,
   212→      ]);
   213→      return true;
   214→    }
   215→    return false;
   216→  }

   258→  function validate(): string | null {
   259→    if (customerName.trim().length < 2) return "نام مشتری را کامل وارد کنید";
   260→    if (!phoneOk) return "شماره موبایل مشتری معتبر نیست (۰۹xxxxxxxxx)";
   261→    if (!cardOk) return "شماره کارت بانکی مشتری معتبر نیست";
   262→    if (!purchaseAt) return "زمان خرید مشتری را انتخاب کنید";
   263→    if (!reason) return "دلیل مرجوعی را انتخاب کنید";
   264→    if (items.length === 0) return "حداقل یک کالا به فهرست اضافه کنید";
   265→    for (const i of items) {
   266→      if (!i.name.trim()) return "نام یکی از اقلام خالی است";
   267→      if (i.quantity < 1) return `تعداد «${i.name}» باید حداقل ۱ باشد`;
   268→      if (i.unitPrice <= 0) return `قیمت «${i.name}» را وارد کنید`;
   269→      if (!i.barcode || digitsOnly(i.barcode).length < 4)
   270→        return `بارکد «${i.name}» خالی است — بارکد را وارد کنید یا دکمه تولید بارکد داخلی را بزنید`;
   271→    }
   272→    // منطق تسویه
   273→    if (exchangeValue > total && transferAmount > 0)
   274→      return "در حالت بدهی مشتری (جایگزین بیشتر از مرجوعی)، مبلغ قابل انتقال باید صفر باشد";
   275→    if (exchangeValue <= total && transferAmount > total - exchangeValue)
   276→      return "مبلغ قابل انتقال نمی‌تواند از «جمع اقلام منهای ارزش جایگزین» بیشتر باشد";
   277→    if (noteRequired && transferNote.trim().length < 3)
   278→      return "دلیل اختلاف مبلغ انتقالی با جمع اقلام را در توضیح تسویه بنویسید";
   279→    return null;
   280→  }

   282→  async function handleSubmit(e?: React.FormEvent) {
   283→    e?.preventDefault();
   284→    if (submitting) return;
   285→    const err = validate();
   286→    if (err) {
   287→      toast.error(err, { description: "لطفاً فرم را کامل کنید" });
   288→      return;
   289→    }
   290→    setSubmitting(true);
   291→    try {
   292→      const res = await fetch("/api/returns", {
   293→        method: "POST",
   294→        headers: { "Content-Type": "application/json" },
   295→        body: JSON.stringify({
   296→          customerName: customerName.trim(),
   297→          customerPhone: digitsOnly(customerPhone),
   298→          customerCardNumber: digitsOnly(customerCard),
   299→          purchaseAt: purchaseAt?.toISOString(),
   300→          reason: reason === "سایر" ? note.trim() || "سایر" : reason,
   301→          note: note.trim() || null,
   302→          exchangeValue,
   303→          transferAmount,
   304→          transferNote: transferNote.trim() || null,
   305→          items: items.map((i) => ({
   306→            productId: i.productId,
   307→            name: i.name.trim(),
   308→            barcode: digitsOnly(i.barcode ?? "") || null,
   309→            code: i.code,
   310→            quantity: i.quantity,
   311→            unitPrice: i.unitPrice,
   312→          })),
   313→        }),
   314→      });
   315→      const data = await res.json();
   316→      if (!res.ok) {
   317→        toast.error(data.error || "ثبت مرجوعی ناموفق بود");
   318→        return;
   319→      }
   320→      setSuccess({
   321→        id: data.id,
   322→        code: data.code,
   323→        total: data.totalAmount,
   324→        transfer: data.transferAmount ?? data.totalAmount,
   325→        exchange: data.exchangeValue ?? 0,
   326→        settlementType: (data.settlementType ?? "full") as SettlementType,
   327→        count: data.itemCount,
   328→        at: new Date(data.submittedAt),
   329→      });
   330→      setTodayCount((c) => c + 1);
   331→      toast.success(`مرجوعی ${data.code} با موفقیت ثبت شد 🌿`, {
   332→        description: "به آقای نوروزی اطلاع داده شد تا وجه را انتقال دهد",
   333→      });
   334→    } catch {
   335→      toast.error("ارتباط با سرور برقرار نشد");
   336→    } finally {
   337→      setSubmitting(false);
   338→    }
   339→  }
