    18→/* ---------- بارکد SVG (EAN-13 استاندارد یا CODE128 برای سایر مقادیر) ---------- */
    19→function BarcodeSvg({
    20→  value,
    21→  height = 30,
    22→  width = 1.6,
    23→  fontSize = 12,
    24→  className = "",
    25→}: {
    26→  value: string;
    27→  height?: number;
    28→  width?: number;
    29→  fontSize?: number;
    30→  className?: string;
    31→}) {
    32→  const ref = useRef<SVGSVGElement>(null);
    33→
    34→  useEffect(() => {
    35→    if (!ref.current || !value) return;
    36→    const isEan13 = /^\d{13}$/.test(value) && isValidEan13(value);
    37→    try {
    38→      JsBarcode(ref.current, value, {
    39→        format: isEan13 ? "EAN13" : "CODE128",
    40→        height,
    41→        width,
    42→        fontSize,
    43→        margin: 2,
    44→        lineColor: "#111111",
    45→        background: "transparent",
    46→        displayValue: true,
    47→        font: "monospace",
    48→        fontOptions: "bold",
    49→        textAlign: "center",
    50→        textPosition: "bottom",
    51→        textMargin: 1,
    52→      });
    53→    } catch {
    54→      // مقدار نامعتبر برای بارکد → نمایش متن خام
    55→      if (ref.current) ref.current.innerHTML = "";
    56→    }
    57→  }, [value, height, width, fontSize]);
    58→
    59→  return <svg ref={ref} className={className} aria-label={`بارکد ${value}`} />;
    60→}

   279→/* ---------- دیالوگ رسید ---------- */
   280→export function ReceiptDialog({
   281→  open,
   282→  onOpenChange,
   283→  detail,
   284→}: {
   285→  open: boolean;
   286→  onOpenChange: (o: boolean) => void;
   287→  detail: ReturnDetailDTO | null;
   288→}) {
   289→  const [settings, setSettings] = useState<StoreSettings>(DEFAULT_SETTINGS);
   290→  // قالب چاپ: برگه A4 یا نوار حرارتی ۸۰ میلی‌متری — انتخاب آخرین بار ذخیره می‌شود
   291→  const [format, setFormat] = useState<"a4" | "thermal">(() => {
   292→    if (typeof window === "undefined") return "a4";
   293→    try {
   294→      return window.localStorage.getItem("hz:receipt-format") === "thermal" ? "thermal" : "a4";
   295→    } catch {
   296→      return "a4";
   297→    }
   298→  });
   299→
   300→  function chooseFormat(f: "a4" | "thermal") {
   301→    setFormat(f);
   302→    try {
   303→      window.localStorage.setItem("hz:receipt-format", f);
   304→    } catch {
   305→      /* حافظه در دسترس نیست */
   306→    }
   307→  }
   308→
   309→  // تنظیمات فروشگاه برای سربرگ/پابرگ رسید
   310→  useEffect(() => {
   311→    if (!open) return;
   312→    let cancelled = false;
   313→    fetch("/api/admin/settings")
   314→      .then((r) => (r.ok ? r.json() : null))
   315→      .then((d) => {
   316→        if (!cancelled && d && typeof d === "object") setSettings({ ...DEFAULT_SETTINGS, ...d });
   317→      })
   318→      .catch(() => {});
   319→    return () => {
   320→      cancelled = true;
   321→    };
   322→  }, [open]);
   323→
   324→  const paid = detail?.status === "paid" || detail?.status === "approved";
   325→  const settlement =
   326→    detail && (detail.transferAmount !== detail.totalAmount || detail.exchangeValue > 0)
   327→      ? {
   328→          T: detail.totalAmount,
   329→          E: detail.exchangeValue,
   330→          X: detail.transferAmount,
   331→          debt: detail.exchangeValue > detail.totalAmount ? detail.exchangeValue - detail.totalAmount : 0,
   332→          credit:
   333→            detail.transferAmount + detail.exchangeValue < detail.totalAmount
   334→              ? detail.totalAmount - detail.exchangeValue - detail.transferAmount
   335→              : 0,
   336→        }
   337→      : null;
   338→
   339→  // ---- اندازهٔ صفحهٔ چاپ حرارتی: به‌اندازهٔ محتوای واقعی (نه ارتفاع ثابت) ----
   340→  // قبلاً @page 80mm×۳۲۷mm بود → کاغذ حرارتی صفحهٔ ثابت ۳۲۷ میلی‌متری چاپ می‌کرد و
   341→  // تا پایان رول کاغذ سفید هدر می‌داد. حالا ارتفاع از محتوای رندرشده اندازه‌گیری می‌شود:
   342→  // چاپ از بالای رسید شروع و دقیقاً جایی که محتوا تمام می‌شود قطع می‌شود.
   343→  const thermalStyleRef = useRef<HTMLStyleElement>(null);
   344→  const thermalPaperRef = useRef<HTMLDivElement>(null);
   345→
   346→  function syncThermalPageSize() {
   347→    if (format !== "thermal") return;
   348→    const paper = thermalPaperRef.current;
   349→    const styleEl = thermalStyleRef.current;
   350→    if (!paper || !styleEl) return;
   351→    // تبدیل پیکسل CSS به میلی‌متر (۹۶ پیکسل = ۱ اینچ = ۲۵٫۴ میلی‌متر)
   352→    const px = Math.ceil(paper.getBoundingClientRect().height);
   353→    const mm = Math.max(30, Math.round((px * 25.4) / 96 + 3)); // حداقل ۳۰mm + بافر ۳mm برای گرد شدن
   354→    styleEl.textContent = `@media print { @page { size: 80mm ${mm}mm; margin: 0; } }`;
   355→  }
   356→
   357→  // همگام‌سازی پس از رندر (باز شدن دیالوگ / تغییر قالب / بارکدها)
   358→  useEffect(() => {
   359→    if (!open || format !== "thermal") return;
   360→    const t1 = requestAnimationFrame(() => requestAnimationFrame(syncThermalPageSize));
   361→    const t2 = window.setTimeout(syncThermalPageSize, 350); // پس از رندر نهایی بارکدها
   362→    return () => {
   363→      cancelAnimationFrame(t1);
   364→      window.clearTimeout(t2);
   365→    };
   366→  }, [open, format, detail?.id]);
   367→
   368→  function handlePrint() {
   369→    // قبل از چاپ، اندازهٔ صفحه دوباره همگام می‌شود — چاپ حرارتی دقیقاً به‌اندازهٔ محتواست
   370→    syncThermalPageSize();
   371→    requestAnimationFrame(() => requestAnimationFrame(() => window.print()));
   372→  }
   373→
   374→  if (!detail) return null;
   375→  const words = moneyWords(detail.transferAmount);
   376→
   377→  return (
   378→    <Dialog open={open} onOpenChange={onOpenChange}>
   379→      <DialogContent className="receipt-dialog-content w-[calc(100vw-1.5rem)] sm:max-w-4xl max-h-[calc(100dvh-2rem)] overflow-y-auto overflow-x-hidden p-0 gap-0 bg-neutral-100 dark:bg-neutral-200">
   380→        <DialogTitle className="sr-only">رسید مرجوعی {detail.code}</DialogTitle>
   381→        <DialogDescription className="sr-only">
   382→          پیش‌نمایش برگه چاپی مرجوعی {detail.code} — قابل چاپ با بارکدهای قابل اسکن
   383→        </DialogDescription>
   384→
   385→        {/* قالب حرارتی: اندازهٔ صفحه چاپ به‌اندازهٔ محتوای واقعی رسید — پس از رندر اندازه‌گیری می‌شود
   386→            (شروع از بالای رسید و قطع خودکار جایی که محتوا تمام می‌شود — بدون کاغذ سفید اضافه) */}
   387→        {format === "thermal" && (
   388→          <style ref={thermalStyleRef}>{"@media print { @page { size: 80mm 120mm; margin: 0; } }"}</style>
   389→        )}
   390→
   391→        {/* نوار ابزار (چاپ نمی‌شود) */}
   392→        <div className="no-print sticky top-0 z-10 flex flex-wrap items-center gap-2 px-4 py-3 bg-neutral-800 text-white border-b border-neutral-700">
   393→          <Printer className="size-4 text-chart-2 shrink-0" />
   394→          <p className="text-xs font-bold flex-1 min-w-0">
   395→            {format === "a4"
   396→              ? "پیش‌نمایش رسید — برگه A4 با بارکدهای قابل اسکن"
   397→              : "پیش‌نمایش رسید — نوار حرارتی ۸۰ میلی‌متری"}
   398→          </p>
   399→          {/* انتخاب قالب چاپ */}
   400→          <div
   401→            className="flex rounded-lg bg-neutral-900/70 p-0.5 border border-neutral-700"
   402→            role="group"
   403→            aria-label="انتخاب قالب چاپ"
   404→          >
   405→            <button
   406→              type="button"
   407→              onClick={() => chooseFormat("a4")}
   408→              aria-pressed={format === "a4"}
   409→              className={`flex items-center gap-1.5 rounded-md px-2.5 py-1.5 text-[11px] font-bold transition-colors h-9 ${
   410→                format === "a4" ? "bg-chart-2 text-neutral-900" : "text-neutral-400 hover:text-white"
   411→              }`}
   412→            >
   413→              <FileText className="size-3.5" />
   414→              برگه A4
   415→            </button>
   416→            <button
   417→              type="button"
   418→              onClick={() => chooseFormat("thermal")}
   419→              aria-pressed={format === "thermal"}
