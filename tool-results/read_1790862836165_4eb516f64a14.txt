    41→interface MonthOption {
    42→  key: string; // "jy-jm" مثل "1405-7"
    43→  jy: number;
    44→  jm: number;
    45→  label: string; // «مهر ۱۴۰۵»

    48→/** برچسب ماه شمسی: «مهر ۱۴۰۵» */
    49→function jalaliMonthLabel(jy: number, jm: number): string {
    50→  return `${JALALI_MONTHS[jm - 1] ?? ""} ${toFaDigits(jy)}`;
    51→}
    52→
    53→/** n ماه اخیر جلالی، از ماه جاری به عقب */
    54→function lastJalaliMonths(n: number): MonthOption[] {
    55→  const now = toJalali(new Date());
    56→  const opts: MonthOption[] = [];
    57→  let jy = now.jy;
    58→  let jm = now.jm;
    59→  for (let i = 0; i < n; i += 1) {
    60→    opts.push({ key: `${jy}-${jm}`, jy, jm, label: jalaliMonthLabel(jy, jm) });
    61→    jm -= 1;
    62→    if (jm === 0) {
    63→      jm = 12;
    64→      jy -= 1;
    65→    }
    66→  }
    67→  return opts;
    68→}

   444→export function ReturnDetailDialog({
   445→  id,
   446→  onClose,
   447→  user,
   448→  onPaid,
   449→}: {
   450→  id: string | null;
   451→  onClose: () => void;
   452→  user: SessionUser;
   453→  onPaid?: () => void;
   454→}) {
   455→  // الگوی استخراج حالت: بارگذاری از مقایسهٔ شناسهٔ درخواستی با دادهٔ واکشی‌شده به‌دست می‌آید
   456→  const [fetched, setFetched] = useState<{ id: string; data: ReturnDetailDTO | null } | null>(null);
   457→  const [receiptOpen, setReceiptOpen] = useState(false);
   458→  const current = fetched && fetched.id === id ? fetched : null;
   459→  const loading = !!id && !current;
   460→  const detail = current?.data ?? null;
   461→
   462→  useEffect(() => {
   463→    if (!id) return;
   464→    let cancelled = false;
   465→    fetch(`/api/returns/${id}`)
   466→      .then((r) => r.json())
   467→      .then((d) => {
   468→        if (cancelled) return;
   469→        if (d.id) setFetched({ id, data: d });
   470→        else {
   471→          setFetched({ id, data: null });
   472→          toast.error(d.error || "خطا در دریافت جزئیات");
   473→        }
   474→      })
   475→      .catch(() => {
   476→        if (!cancelled) {
   477→          setFetched({ id, data: null });
   478→          toast.error("خطا در دریافت جزئیات");
   479→        }
   480→      });
   481→    return () => {
   482→      cancelled = true;
   483→    };
   484→  }, [id]);
   485→
   486→  // پاک‌سازی جزئیات هنگام بستن پنجره (در رویداد، نه افکت)
   487→  function handleClose() {
   488→    setFetched(null);
   489→    onClose();
   490→  }

   885→export function PaymentForm({
   886→  detail,
   887→  onPaid,
   888→}: {
   889→  detail: ReturnDetailDTO;
   890→  onPaid: (code: string, noTransfer?: boolean) => void;
   891→}) {
   892→  // حالت تسویه بدون انتقال وجه — تبادل کالا / بدهی مشتری / اعتبار
   893→  const noTransfer = detail.transferAmount === 0;
   894→  const [bankName, setBankName] = useState("");
   895→  const [accountNumber, setAccountNumber] = useState("");
   896→  const [cardNumber, setCardNumber] = useState("");
   897→  const [referenceNumber, setReferenceNumber] = useState("");
   898→  const [transferredAt, setTransferredAt] = useState<Date | null>(null);
   899→  const [submitting, setSubmitting] = useState(false);
   900→
   901→  async function submit() {
   902→    if (submitting) return;
   903→    if (!transferredAt) {
   904→      toast.error(noTransfer ? "زمان تسویه را انتخاب کنید" : "زمان انتقال وجه را انتخاب کنید");
   905→      return;
   906→    }
   907→    if (!noTransfer) {
   908→      if (!bankName) {
   909→        toast.error("بانک مبدأ را انتخاب کنید");
   910→        return;
   911→      }
   912→      if (!accountNumber && !digitsOnly(cardNumber) && !referenceNumber) {
   913→        toast.error("حداقل شماره حساب، کارت یا شماره پیگیری را وارد کنید");
   914→        return;
   915→      }
   916→      if (cardNumber && !/^\d{16}$/.test(digitsOnly(cardNumber))) {
   917→        toast.error("شماره کارت مبدأ باید ۱۶ رقم باشد");
   918→        return;
   919→      }
   920→    }
   921→    setSubmitting(true);
   922→    try {
   923→      const res = await fetch(`/api/returns/${detail.id}/payment`, {
   924→        method: "POST",
   925→        headers: { "Content-Type": "application/json" },
   926→        body: JSON.stringify(
   927→          noTransfer
   928→            ? { noTransfer: true, transferredAt: transferredAt.toISOString() }
   929→            : {
   930→                bankName,
   931→                accountNumber: accountNumber.trim() || null,
   932→                cardNumber: digitsOnly(cardNumber) || null,
   933→                referenceNumber: referenceNumber.trim() || null,
   934→                transferredAt: transferredAt.toISOString(),
   935→              }
   936→        ),
   937→      });
   938→      const data = await res.json();
   939→      if (!res.ok) {
   940→        toast.error(data.error || "ثبت پرداخت ناموفق بود");
   941→        return;
   942→      }
   943→      onPaid(data.code, noTransfer);
   944→    } catch {
   945→      toast.error("ارتباط با سرور برقرار نشد");
   946→    } finally {
   947→      setSubmitting(false);
   948→    }
   949→  }
   950→
   951→  const amountDiffers = detail.transferAmount !== detail.totalAmount || detail.exchangeValue > 0;
   952→
   953→  return (
   954→    <section className="rounded-xl border-2 border-primary/30 bg-card p-4 space-y-4">
