    18→/* ---------- بارکد SVG (EAN-13 استاندارد یا CODE128 برای سایر مقادیر) ---------- */
    19→function BarcodeSvg({
    20→  value,
    21→  height = 30,
    22→  width = 1.6,
    23→  fontSize = 12,
    24→  className = "",
    25→}: {
    26→  value: string;
    27→  height?: number;
    28→  width?: number;
    29→  fontSize?: number;
    30→  className?: string;
    31→}) {
    32→  const ref = useRef<SVGSVGElement>(null);
    33→
    34→  useEffect(() => {
    35→    if (!ref.current || !value) return;
    36→    const isEan13 = /^\d{13}$/.test(value) && isValidEan13(value);
    37→    try {
    38→      JsBarcode(ref.current, value, {
    39→        format: isEan13 ? "EAN13" : "CODE128",
    40→        height,
    41→        width,
    42→        fontSize,
    43→        margin: 2,
    44→        lineColor: "#111111",
    45→        background: "transparent",
    46→        displayValue: true,
    47→        font: "monospace",
    48→        fontOptions: "bold",
    49→        textAlign: "center",
    50→        textPosition: "bottom",
    51→        textMargin: 1,
    52→      });
    53→    } catch {
    54→      // مقدار نامعتبر برای بارکد → نمایش متن خام
    55→      if (ref.current) ref.current.innerHTML = "";
    56→    }
    57→  }, [value, height, width, fontSize]);
    58→
    59→  return <svg ref={ref} className={className} aria-label={`بارکد ${value}`} />;
    60→}

   278→export function ReceiptDialog({
   279→  open,
   280→  onOpenChange,
   281→  detail,
   282→}: {
   283→  open: boolean;
   284→  onOpenChange: (o: boolean) => void;
   285→  detail: ReturnDetailDTO | null;
   286→}) {
   287→  const [settings, setSettings] = useState<StoreSettings>(DEFAULT_SETTINGS);
   288→  // قالب چاپ: برگه A4 یا نوار حرارتی ۸۰ میلی‌متری — انتخاب آخرین بار ذخیره می‌شود
   289→  const [format, setFormat] = useState<"a4" | "thermal">(() => {
   290→    if (typeof window === "undefined") return "a4";
   291→    try {
   292→      return window.localStorage.getItem("hz:receipt-format") === "thermal" ? "thermal" : "a4";
   293→    } catch {
   294→      return "a4";
   295→    }
   296→  });
   297→
   298→  function chooseFormat(f: "a4" | "thermal") {
   299→    setFormat(f);
   300→    try {
   301→      window.localStorage.setItem("hz:receipt-format", f);
   302→    } catch {
   303→      /* حافظه در دسترس نیست */
   304→    }
   305→  }
   306→
   307→  // تنظیمات فروشگاه برای سربرگ/پابرگ رسید
   308→  useEffect(() => {
   309→    if (!open) return;
   310→    let cancelled = false;
   311→    fetch("/api/admin/settings")
   312→      .then((r) => (r.ok ? r.json() : null))
   313→      .then((d) => {
   314→        if (!cancelled && d && typeof d === "object") setSettings({ ...DEFAULT_SETTINGS, ...d });
   315→      })
   316→      .catch(() => {});
   317→    return () => {
   318→      cancelled = true;
   319→    };
   320→  }, [open]);
   321→
   322→  const paid = detail?.status === "paid";
   323→  const settlement =
   324→    detail && (detail.transferAmount !== detail.totalAmount || detail.exchangeValue > 0)
   325→      ? {
   326→          T: detail.totalAmount,
   327→          E: detail.exchangeValue,
   328→          X: detail.transferAmount,
   329→          debt: detail.exchangeValue > detail.totalAmount ? detail.exchangeValue - detail.totalAmount : 0,
   330→          credit:
   331→            detail.transferAmount + detail.exchangeValue < detail.totalAmount
   332→              ? detail.totalAmount - detail.exchangeValue - detail.transferAmount
   333→              : 0,
   334→        }
   335→      : null;
   336→
   337→  function handlePrint() {
   338→    window.print();
   339→  }
   340→
   341→  if (!detail) return null;
   342→  const words = moneyWords(detail.transferAmount);
   343→
   344→  return (

   346→      <DialogContent className="receipt-dialog-content w-[calc(100vw-1.5rem)] sm:max-w-4xl max-h-[calc(100dvh-2rem)] overflow-y-auto overflow-x-hidden p-0 gap-0 bg-neutral-100 dark:bg-neutral-200">

   349→          پیش‌نمایش برگه چاپی مرجوعی {detail.code} — قابل چاپ با بارکدهای قابل اسکن

   352→        {/* قالب حرارتی: اندازهٔ صفحهٔ چاپ ۸۰×۳۲۷ میلی‌متر (نوار پیوسته بلند) — این تگ بعد از globals.css می‌آید و برنده می‌شود */}

   439→            <div className="flex items-center justify-between gap-3 py-3 border-b border-dashed border-neutral-300">
   440→              <h3 className="text-base font-extrabold text-neutral-900">رسید ثبت مرجوعی کالا</h3>
   441→              <div className="flex items-center gap-2 text-[10px] font-bold">
   442→                {paid ? (
   443→                  <span className="inline-flex items-center gap-1 border border-neutral-900 px-2 py-0.5 rounded">
   444→                    <CheckCircle2 className="size-3" />
   445→                    تسویه‌شده (پرداخت/بایگانی)
   446→                  </span>
   447→                ) : (
   448→                  <span className="inline-flex items-center gap-1 border-2 border-neutral-900 px-2 py-0.5 rounded">
   449→                    <Hourglass className="size-3" />
   450→                    در انتظار انتقال وجه
   451→                  </span>
   452→                )}
   453→              </div>
   454→            </div>
   455→
   456→            {/* اطلاعات مشتری و ثبت */}
   457→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-x-6 py-3 border-b border-dashed border-neutral-300">
   458→              <div>
   459→                <p className="text-[11px] font-extrabold border-b border-neutral-300 pb-1 mb-1.5">مشخصات مشتری</p>
   460→                <Field label="نام و نام خانوادگی" value={detail.customerName} strong />
   461→                <Field label="موبایل" value={detail.customerPhone} ltr />
   462→                <Field label="کارت بانکی" value={formatCardNumber(detail.customerCardNumber)} ltr />
   463→                <Field label="زمان خرید" value={formatJalaliDate(new Date(detail.purchaseAt))} />
   464→              </div>
   465→              <div>
   466→                <p className="text-[11px] font-extrabold border-b border-neutral-300 pb-1 mb-1.5">اطلاعات ثبت</p>
   467→                <Field label="صندوقدار ثبت‌کننده" value={detail.cashier.fullName} strong />
   468→                <Field label="زمان ثبت" value={formatJalaliFull(new Date(detail.submittedAt))} />
   469→                <Field label="دلیل مرجوعی" value={detail.reason} />
   470→                {detail.note && <Field label="توضیحات" value={detail.note} />}
   471→              </div>
   472→            </div>
   473→
   474→            {/* اقلام + بارکد */}
   475→            <div className="py-3">
   476→              <p className="text-[11px] font-extrabold border-b border-neutral-300 pb-1 mb-2">
   477→                اقلام مرجوعی ({detail.items.length} قلم)
   478→              </p>
   479→              <table className="w-full text-[10px] border-collapse">
   480→                <thead>
   481→                  <tr className="bg-neutral-100 text-neutral-700">
   482→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-7 font-bold">#</th>
   483→                    <th className="border border-neutral-300 px-1.5 py-1.5 font-bold text-right">نام کالا</th>
   484→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-20 font-bold">کد</th>
   485→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-36 font-bold">بارکد</th>
   486→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-10 font-bold">تعداد</th>
   487→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-24 font-bold">قیمت واحد (ریال)</th>
   488→                    <th className="border border-neutral-300 px-1.5 py-1.5 w-24 font-bold">مبلغ کل (ریال)</th>
   489→                  </tr>
   490→                </thead>
   491→                <tbody>
   492→                  {detail.items.map((it, i) => (
   493→                    <tr key={it.id} className={i % 2 === 1 ? "bg-neutral-50" : ""}>
   494→                      <td className="border border-neutral-300 px-1.5 py-2 text-center num">{i + 1}</td>
   495→                      <td className="border border-neutral-300 px-1.5 py-2 font-semibold leading-4">{it.name}</td>
   496→                      <td className="border border-neutral-300 px-1.5 py-2 text-center num" dir="ltr">
   497→                        {it.code || "—"}
   498→                      </td>
   499→                      <td className="border border-neutral-300 px-1.5 py-1.5">
   500→                        <div className="flex items-center justify-center min-h-8">
   501→                          {it.barcode ? (
   502→                            <BarcodeSvg value={it.barcode} height={22} width={1.1} fontSize={9} />
   503→                          ) : (
   504→                            <span className="text-neutral-400">—</span>
   505→                          )}
   506→                        </div>
   507→                      </td>
   508→                      <td className="border border-neutral-300 px-1.5 py-2 text-center num font-bold">{it.quantity}</td>
   509→                      <td className="border border-neutral-300 px-1.5 py-2 text-center num">{formatMoney(it.unitPrice)}</td>
   510→                      <td className="border border-neutral-300 px-1.5 py-2 text-center num font-bold">{formatMoney(it.lineTotal)}</td>
   511→                    </tr>
   512→                  ))}
   513→                </tbody>
   514→              </table>
   515→            </div>
   516→
   517→            {/* تسویه با مشتری */}
   518→            <div className="grid grid-cols-1 sm:grid-cols-[1fr_auto] gap-4 items-start">
   519→              <div className="space-y-2.5">
   520→                {settlement ? (
   521→                  <div className="border border-neutral-400 rounded p-3">
   522→                    <p className="text-[11px] font-extrabold border-b border-neutral-300 pb-1 mb-1.5 flex items-center justify-between">
   523→                      <span>تسویه با مشتری ({SETTLEMENT_META[detail.settlementType]?.label ?? detail.settlementType})</span>
   524→                    </p>
   525→                    <Field label="جمع کل اقلام" value={`${formatMoney(settlement.T)} ریال`} />
   526→                    {settlement.E > 0 && (
   527→                      <Field label="ارزش کالای جایگزین (−)" value={`${formatMoney(settlement.E)} ریال`} />
   528→                    )}
   529→                    <Field label="مبلغ قابل انتقال به حساب مشتری" value={`${formatMoney(settlement.X)} ریال`} strong />
   530→                    {settlement.debt > 0 && (
   531→                      <Field label="بدهی مشتری به فروشگاه" value={`${formatMoney(settlement.debt)} ریال`} strong />
   532→                    )}
   533→                    {settlement.credit > 0 && (
   534→                      <Field label="اعتبار باقی‌مانده مشتری" value={`${formatMoney(settlement.credit)} ریال`} />
   535→                    )}
   536→                    {detail.transferNote && (
   537→                      <p className="mt-2 pt-2 border-t border-dashed border-neutral-300 text-[10px] leading-5">
   538→                        <span className="font-bold">دلیل اختلاف: </span>
   539→                        {detail.transferNote}
   540→                      </p>
   541→                    )}
   542→                  </div>
   543→                ) : (
   544→                  <div className="border border-neutral-300 rounded p-3">
   545→                    <Field label="تسویه" value="انتقال کامل وجه به حساب مشتری" />
   546→                  </div>
   547→                )}
   548→
   549→                {/* اطلاعات پرداخت */}
   550→                <div className="border border-neutral-400 rounded p-3">
   551→                  <p className="text-[11px] font-extrabold border-b border-neutral-300 pb-1 mb-1.5">اطلاعات بازگشت وجه</p>
   552→                  {paid && detail.payment ? (
   553→                    <>
   554→                      <Field label="وضعیت" value="تسویه و بایگانی شد" strong />
   555→                      <Field label="بانک" value={detail.payment.bankName} />
   556→                      {detail.payment.referenceNumber && (
   557→                        <Field label="شماره پیگیری انتقال" value={detail.payment.referenceNumber} ltr />
   558→                      )}
   559→                      {detail.payment.accountNumber && (
   560→                        <Field label="شماره حساب مبدأ" value={detail.payment.accountNumber} ltr />
   561→                      )}
   562→                      {detail.payment.cardNumber && (
   563→                        <Field label="کارت مبدأ" value={formatCardNumber(detail.payment.cardNumber)} ltr />
   564→                      )}
   565→                      <Field label="زمان انتقال/تسویه" value={formatJalaliFull(new Date(detail.payment.transferredAt))} />
   566→                      <Field label="ثبت‌کننده" value={detail.payment.recordedBy.fullName} />
   567→                    </>
   568→                  ) : (
   569→                    <p className="text-[10px] leading-5 text-neutral-700">
   570→                      وجه این مرجوعی هنوز منتقل نشده است — پس از تأیید مدیریت ({settings.managerName})، مبلغ
   571→                      قابل انتقال به کارت بانکی مشتری واریز می‌شود.
   572→                    </p>
   573→                  )}
   574→                </div>
   575→              </div>
   576→
   577→              {/* مبلغ نهایی */}
   578→              <div className="sm:w-56 border-2 border-neutral-900 rounded p-4 text-center">
   579→                <p className="text-[10px] font-bold text-neutral-600 mb-1">
   580→                  {paid ? "مبلغ انتقالی نهایی" : "مبلغ قابل انتقال"}
   581→                </p>
   582→                <p className="text-xl font-extrabold num text-neutral-900 leading-8">
   583→                  {formatMoney(detail.transferAmount)}
   584→                </p>
   585→                <p className="text-[10px] text-neutral-600 num mb-1">ریال</p>
   586→                <p className="text-[9px] text-neutral-500 leading-4 border-t border-dashed border-neutral-300 pt-1.5">
   587→                  {words.rial}
   588→                  <br />
   589→                  {words.toman}
   590→                </p>
   591→              </div>
   592→            </div>
   593→
   594→            {/* امضاها */}
   595→            <div className="grid grid-cols-3 gap-6 mt-8 pt-2">
   596→              {["امضای مشتری", "امضای صندوقدار", `تأیید ${settings.managerName}`].map((s) => (
   597→                <div key={s} className="text-center">
   598→                  <div className="border-t border-dotted border-neutral-500 pt-1.5 mx-3" style={{ height: 34 }} />
   599→                  <p className="text-[9px] text-neutral-600">{s}</p>
   600→                </div>
   601→              ))}
   602→            </div>
   603→
   604→            {/* پابرگ */}
   605→            <div className="mt-6 pt-3 border-t border-neutral-300 text-center space-y-1">
   606→              <p className="text-[10px] font-semibold text-neutral-800 leading-5">{settings.receiptFooter}</p>
   607→              <p className="text-[8px] text-neutral-500 leading-4 flex items-center justify-center gap-1 flex-wrap">
   608→                <Leaf className="size-2.5 inline" aria-hidden="true" />
   609→                توسعه‌دهنده سامانه: {settings.developerCredit}
   610→                <span className="mx-1">—</span>
   611→                <span className="num">چاپ: {formatJalaliFull(new Date())}</span>
   612→              </p>
   613→            </div>
   614→          </div>
   615→          ) : (
   616→            <ThermalReceipt detail={detail} settlement={settlement} words={words} settings={settings} paid={paid} />
   617→          )}
   618→        </div>
   619→      </DialogContent>
   620→    </Dialog>
   621→  );
   622→}
