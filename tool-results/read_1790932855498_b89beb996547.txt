    77→/** برچسب کوتاه شمسی محور افقی: «۱۲ مهر» */
    78→function shortJalali(isoDate: string): string {
    79→  const [y, m, d] = isoDate.split("-").map((p) => parseInt(p, 10));
    80→  const j = toJalali(new Date(y || 2025, (m || 1) - 1, d || 1));
    81→  return `${toFaDigits(j.jd)} ${JALALI_MONTHS[j.jm - 1]}`;
    82→}
    83→
    84→/** کوتاه‌سازی نام کالا برای برچسب محور عمودی */
    85→function truncateName(name: string, max = 24): string {
    86→  return name.length > max ? `${name.slice(0, max - 1)}…` : name;
    87→}
