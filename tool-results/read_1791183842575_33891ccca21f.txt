   108→/** کوچک‌سازی تصویر سمت کلاینت (۲۵۶×۲۵۶ JPEG) و تبدیل به dataURL */
   109→function resizeImageFile(file: File): Promise<string> {
   110→  return new Promise((resolve, reject) => {
   111→    if (!file.type.startsWith("image/")) {
   112→      reject(new Error("فایل انتخاب‌شده تصویر نیست"));
   113→      return;
   114→    }
   115→    if (file.size > 15 * 1024 * 1024) {
   116→      reject(new Error("حجم تصویر نباید بیشتر از ۱۵ مگابایت باشد"));
   117→      return;
   118→    }
   119→    const reader = new FileReader();
   120→    reader.onerror = () => reject(new Error("خواندن فایل ناموفق بود"));
   121→    reader.onload = () => {
   122→      const img = new Image();
   123→      img.onerror = () => reject(new Error("تصویر قابل خواندن نیست"));
   124→      img.onload = () => {
   125→        const SIDE = 256;
   126→        const canvas = document.createElement("canvas");
   127→        // برش مربعی از مرکز تصویر
   128→        const side = Math.min(img.width, img.height);
   129→        const sx = (img.width - side) / 2;
   130→        const sy = (img.height - side) / 2;
   131→        canvas.width = SIDE;
   132→        canvas.height = SIDE;
   133→        const ctx = canvas.getContext("2d");
   134→        if (!ctx) {
   135→          reject(new Error("پردازش تصویر ناموفق بود"));
   136→          return;
   137→        }
   138→        ctx.drawImage(img, sx, sy, side, side, 0, 0, SIDE, SIDE);
   139→        resolve(canvas.toDataURL("image/jpeg", 0.85));
   140→      };
   141→      img.src = String(reader.result);
   142→    };
   143→    reader.readAsDataURL(file);
   144→  });
   145→}

   170→    savedAvatar?.type === "vector" ? "vector" : savedAvatar?.type === "photo" ? "photo" : "initials"
   171→  );

   173→  const [avatarPhoto, setAvatarPhoto] = useState(savedAvatar?.photo ?? "");
   174→  // آواتار کارتونی دوبعدی — پیکربندی زنده
   175→  const [vectorConfig, setVectorConfig] = useState<VectorAvatarConfig>(
   176→    savedAvatar?.type === "vector" && savedAvatar.config
   177→      ? savedAvatar.config
   178→      : DEFAULT_VECTOR_CONFIG
   179→  );
   180→  const [creatorCategory, setCreatorCategory] = useState<CreatorCategory>("skin");
   181→  const [savingAvatar, setSavingAvatar] = useState(false);
   182→  const [processingPhoto, setProcessingPhoto] = useState(false);
   183→
   184→  // ---- تغییر گذرواژه ----
   185→  const [currentPassword, setCurrentPassword] = useState("");
   186→  const [newPassword, setNewPassword] = useState("");
   187→  const [repeatPassword, setRepeatPassword] = useState("");
   188→  const [showCurrent, setShowCurrent] = useState(false);
   189→  const [showNew, setShowNew] = useState(false);
   190→  const [showRepeat, setShowRepeat] = useState(false);
   191→  const [passErrors, setPassErrors] = useState<PasswordErrors>({});
   192→  const [changingPassword, setChangingPassword] = useState(false);
   193→
   194→  // بازکردن پنجره حساب من — بازنشانی فرم‌ها و دریافت مشخصات فعلی

   199→    setProfileErrors({});
   200→    const av = parseAvatar(user.avatar);
   201→    setAvatarMode(
   202→      av?.type === "vector" ? "vector" : av?.type === "photo" ? "photo" : "initials"
   203→    );
   204→    setAvatarColor(av?.color ?? AVATAR_COLORS[0]);
   205→    setAvatarPhoto(av?.photo ?? "");
   206→    setVectorConfig(
   207→      av?.type === "vector" && av.config ? av.config : DEFAULT_VECTOR_CONFIG
   208→    );
   209→    setCreatorCategory("skin");
   210→    setCurrentPassword("");
   211→    setNewPassword("");
   212→    setRepeatPassword("");
   213→    setShowCurrent(false);
   214→    setShowNew(false);
   215→    setShowRepeat(false);
   216→    setPassErrors({});
   217→    setDialogOpen(true);
   218→
   219→    fetch("/api/auth/profile")
   220→      .then((r) => r.json())
   221→      .then((d) => {
   222→        if (d?.user) {
   223→          setPhone(d.user.phone ?? "");
   224→          setInitialPhone(d.user.phone ?? "");
   225→        }
   226→      })
   227→      .catch(() => setInitialPhone(""));
   228→  };
   229→
   230→  // ---- ذخیره مشخصات شخصی ----
   231→  const saveProfile = async () => {
   232→    if (savingProfile) return;
   233→    const name = fullName.trim();
   234→    const ph = digitsOnly(toEnDigits(phone));
   235→
   236→    const e: { fullName?: string; phone?: string } = {};
   237→    if (name.length < 3) e.fullName = "نام کامل حداقل ۳ نویسه باشد";
   238→    if (ph && !/^09\d{9}$/.test(ph)) e.phone = "شماره موبایل باید ۱۱ رقم و با ۰۹ شروع شود";
   239→    setProfileErrors(e);
   240→    if (Object.keys(e).length > 0) return;
   241→
   242→    // تنها فیلدهای تغییر‌یافته ارسال می‌شود
   243→    const payload: Record<string, string> = {};
   244→    if (name !== user.fullName) payload.fullName = name;
   245→    // تا پیش از دریافت شماره فعلی، تغییر شماره ارسال نمی‌شود
   246→    if (initialPhone !== null && ph !== initialPhone) payload.phone = ph;
   247→
   248→    if (Object.keys(payload).length === 0) {
   249→      toast.info("تغییری برای ذخیره وجود ندارد");
   250→      return;
   251→    }
   252→
   253→    setSavingProfile(true);
   254→    try {
   255→      const res = await fetch("/api/auth/profile", {
   256→        method: "PATCH",
   257→        headers: { "Content-Type": "application/json" },
   258→        body: JSON.stringify(payload),
   259→      });
   260→      const data = await res.json();
   261→      if (res.ok) {
   262→        toast.success("مشخصات شما به‌روزرسانی شد");
   263→        setInitialPhone(ph);
   264→        onUserUpdated?.({
   265→          id: data.user.id,
   266→          username: data.user.username,
   267→          fullName: data.user.fullName,
   268→          role: data.user.role,
   269→          phone: data.user.phone ?? null,
   270→          avatar: data.user.avatar ?? null,
   271→        });
   272→      } else {
   273→        toast.error(data.error || "خطا در ذخیره مشخصات");
   274→      }
   275→    } catch {
   276→      toast.error("خطا در ذخیره مشخصات");
   277→    } finally {
   278→      setSavingProfile(false);
   279→    }

   314→        method: "PATCH",

   316→        body: JSON.stringify({ avatar: avatarJson }),
   317→      });

   319→      if (res.ok) {
   320→        toast.success("آواتار شما به‌روزرسانی شد");
   321→        onUserUpdated?.({
   322→          id: user.id,
   323→          username: user.username,
   324→          fullName: user.fullName,
   325→          role: user.role,
   326→          avatar: data.user.avatar ?? null,
   327→        });
   328→      } else {
   329→        toast.error(data.error || "خطا در ذخیره آواتار");
   330→      }
   331→    } catch {
   332→      toast.error("خطا در ذخیره آواتار");
   333→    } finally {
   334→      setSavingAvatar(false);
   335→    }
   336→  }

   348→    setChangingPassword(true);

   350→      const res = await fetch("/api/auth/password", {
   351→        method: "POST",
   352→        headers: { "Content-Type": "application/json" },
   353→        body: JSON.stringify({ currentPassword, newPassword }),
   354→      });
   355→      const data = await res.json();
   356→      if (res.ok) {
   357→        toast.success("گذرواژه با موفقیت تغییر کرد");
   358→        setCurrentPassword("");
   359→        setNewPassword("");

   362→        setShowNew(false);
   363→        setShowRepeat(false);
   364→        setPassErrors({});
   365→      } else {
   366→        toast.error(data.error || "خطا در تغییر گذرواژه");

   368→    } catch {
   369→      toast.error("خطا در تغییر گذرواژه");

   378→        <DropdownMenuTrigger asChild>
   379→          <Button

   381→            className="size-11 sm:size-9 shrink-0 rounded-full p-0 hover:bg-accent"
   382→            aria-label="حساب کاربری من"
   383→            title="حساب کاربری من"
   384→          >
   385→            <AvatarBadge fullName={user.fullName} avatar={user.avatar} size="md" />
   386→          </Button>
   387→        </DropdownMenuTrigger>
   388→        <DropdownMenuContent align="end" className="min-w-56">
   389→          {/* سربرگ: مشخصات کاربر جاری */}
   390→          <DropdownMenuLabel className="font-normal p-0">
   391→            <div className="flex items-center gap-3 px-2 py-2">
   392→              <AvatarBadge fullName={user.fullName} avatar={user.avatar} size="md" />
   393→              <div className="min-w-0 flex flex-col gap-0.5">
   394→                <p className="text-sm font-bold truncate max-w-40">{user.fullName}</p>
