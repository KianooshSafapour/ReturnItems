   235→                size="icon"
   236→                className="size-8"
   237→                disabled={page >= pages}

   256→        }}

   262→// ---------- پنجره جزئیات مرجوعی ----------
   263→export function ReturnDetailDialog({
   264→  id,
   265→  onClose,
   266→  user,
   267→  onPaid,
   268→}: {
   269→  id: string | null;
   270→  onClose: () => void;
   271→  user: SessionUser;
   272→  onPaid?: () => void;
   273→}) {
   274→  // الگوی استخراج حالت: بارگذاری از مقایسهٔ شناسهٔ درخواستی با دادهٔ واکشی‌شده به‌دست می‌آید
   275→  const [fetched, setFetched] = useState<{ id: string; data: ReturnDetailDTO | null } | null>(null);
   276→  const current = fetched && fetched.id === id ? fetched : null;
   277→  const loading = !!id && !current;
   278→  const detail = current?.data ?? null;
   279→
   280→  useEffect(() => {
   281→    if (!id) return;
   282→    let cancelled = false;
   283→    fetch(`/api/returns/${id}`)
   284→      .then((r) => r.json())
   285→      .then((d) => {
   286→        if (cancelled) return;
   287→        if (d.id) setFetched({ id, data: d });
   288→        else {
   289→          setFetched({ id, data: null });
   290→          toast.error(d.error || "خطا در دریافت جزئیات");
   291→        }
   292→      })
   293→      .catch(() => {
   294→        if (!cancelled) {
   295→          setFetched({ id, data: null });
   296→          toast.error("خطا در دریافت جزئیات");
   297→        }
   298→      });
   299→    return () => {
   300→      cancelled = true;
   301→    };
   302→  }, [id]);
   303→
   304→  // پاک‌سازی جزئیات هنگام بستن پنجره (در رویداد، نه افکت)
   305→  function handleClose() {
   306→    setFetched(null);
   307→    onClose();
   308→  }
   309→
   310→  return (
   311→    <Dialog open={!!id} onOpenChange={(o) => !o && handleClose()}>
   312→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto p-0" dir="rtl">
   313→        {loading || !detail ? (
   314→          <div className="p-10 flex flex-col items-center gap-4">
   315→            <DialogHeader className="sr-only">
   316→              <DialogTitle>جزئیات مرجوعی</DialogTitle>
   317→              <DialogDescription>در حال دریافت جزئیات از سرور</DialogDescription>
   318→            </DialogHeader>
   319→            <Loader2 className="size-8 animate-spin text-primary" />
   320→            <p className="text-sm text-muted-foreground">در حال دریافت جزئیات…</p>
   321→          </div>
   322→        ) : (
   323→          <div>
   324→            <DialogHeader className="px-6 pt-6 pb-0">
   325→              <DialogTitle className="flex items-center gap-2 flex-wrap">
   326→                <span className="num text-primary" dir="ltr">
   327→                  {detail.code}
   328→                </span>
   329→                {detail.status === "pending" ? (
   330→                  <Badge className="bg-chart-2/15 text-chart-2 border-chart-2/30 gap-1">
   331→                    <Hourglass className="size-3" /> در انتظار پرداخت
   332→                  </Badge>
   333→                ) : (
   334→                  <Badge className="bg-primary/15 text-primary border-primary/30 gap-1">
   335→                    <CheckCircle2 className="size-3" /> پرداخت و بایگانی شد
   336→                  </Badge>
   337→                )}
   338→              </DialogTitle>
   339→              <DialogDescription className="sr-only">
   340→                جزئیات کامل مرجوعی، اقلام و اطلاعات پرداخت
   341→              </DialogDescription>
   342→            </DialogHeader>
   343→
   344→            <div className="p-6 space-y-6">
   345→              {/* اطلاعات مشتری */}
   346→              <section className="grid sm:grid-cols-2 gap-3 text-sm">
   347→                <InfoRow icon={<UserRound className="size-4" />} label="مشتری" value={detail.customerName} />
   348→                <InfoRow
   349→                  icon={<Phone className="size-4" />}
   350→                  label="موبایل"
   351→                  value={toFaDigits(detail.customerPhone)}
   352→                  ltr
   353→                />
   354→                <InfoRow
   355→                  icon={<CreditCard className="size-4" />}
   356→                  label="کارت بانکی مشتری"
   357→                  value={formatCardNumber(detail.customerCardNumber)}
   358→                  ltr
   359→                />
   360→                <InfoRow
   361→                  icon={<CalendarClock className="size-4" />}
   362→                  label="زمان خرید"
   363→                  value={formatJalaliFull(new Date(detail.purchaseAt))}

   545→export function PaymentForm({
   546→  detail,
   547→  onPaid,
   548→}: {
   549→  detail: ReturnDetailDTO;
   550→  onPaid: (code: string) => void;
   551→}) {
   552→  const [bankName, setBankName] = useState("");
   553→  const [accountNumber, setAccountNumber] = useState("");
   554→  const [cardNumber, setCardNumber] = useState("");
   555→  const [referenceNumber, setReferenceNumber] = useState("");
   556→  const [transferredAt, setTransferredAt] = useState<Date | null>(null);
   557→  const [submitting, setSubmitting] = useState(false);
   558→
   559→  async function submit() {
   560→    if (submitting) return;
   561→    if (!bankName) {
   562→      toast.error("بانک مبدأ را انتخاب کنید");
   563→      return;
   564→    }
   565→    if (!accountNumber && !digitsOnly(cardNumber) && !referenceNumber) {
   566→      toast.error("حداقل شماره حساب، کارت یا شماره پیگیری را وارد کنید");
   567→      return;
   568→    }
   569→    if (cardNumber && !/^\d{16}$/.test(digitsOnly(cardNumber))) {
   570→      toast.error("شماره کارت مبدأ باید ۱۶ رقم باشد");
   571→      return;
   572→    }
   573→    if (!transferredAt) {
   574→      toast.error("زمان انتقال وجه را انتخاب کنید");
   575→      return;
   576→    }
   577→    setSubmitting(true);
   578→    try {
   579→      const res = await fetch(`/api/returns/${detail.id}/payment`, {
   580→        method: "POST",
   581→        headers: { "Content-Type": "application/json" },
   582→        body: JSON.stringify({
   583→          bankName,
   584→          accountNumber: accountNumber.trim() || null,
   585→          cardNumber: digitsOnly(cardNumber) || null,
   586→          referenceNumber: referenceNumber.trim() || null,
   587→          transferredAt: transferredAt.toISOString(),
   588→        }),
   589→      });
   590→      const data = await res.json();
   591→      if (!res.ok) {
   592→        toast.error(data.error || "ثبت پرداخت ناموفق بود");
   593→        return;
   594→      }
   595→      onPaid(data.code);
   596→    } catch {
   597→      toast.error("ارتباط با سرور برقرار نشد");
   598→    } finally {
   599→      setSubmitting(false);
   600→    }
   601→  }
   602→
   603→  return (
   604→    <section className="rounded-xl border-2 border-primary/30 bg-card p-4 space-y-4">
   605→      <div className="flex items-center gap-2">
   606→        <Wallet className="size-4 text-primary" />
   607→        <h4 className="font-bold text-sm">ثبت بازگشت وجه (ویژه مدیر)</h4>
   608→      </div>
   609→      <IlluminatedDivider />
   610→      <div className="rounded-lg bg-primary/5 border border-primary/20 p-3">
   611→        <MoneyDisplay amount={detail.totalAmount} size="md" />
   612→      </div>
   613→      <div className="grid sm:grid-cols-2 gap-3">
   614→        <div className="space-y-1.5">
   615→          <label className="text-xs font-semibold flex items-center gap-1">
   616→            <Landmark className="size-3.5 text-primary" />
   617→            بانک مبدأ (انتقال وجه)
   618→          </label>
   619→          <Select value={bankName} onValueChange={setBankName}>
   620→            <SelectTrigger className="h-10">
   621→              <SelectValue placeholder="انتخاب بانک…" />
   622→            </SelectTrigger>
   623→            <SelectContent className="max-h-72">
   624→              {BANKS.map((b) => (
   625→                <SelectItem key={b} value={b}>
   626→                  {b}
   627→                </SelectItem>
   628→              ))}
   629→            </SelectContent>
   630→          </Select>
   631→        </div>
   632→        <div className="space-y-1.5">
   633→          <label className="text-xs font-semibold">زمان انتقال وجه</label>
   634→          <JalaliDateTimePicker value={transferredAt} onChange={setTransferredAt} placeholder="انتخاب زمان انتقال" />
   635→        </div>
   636→        <div className="space-y-1.5">
   637→          <label className="text-xs font-semibold">شماره حساب مبدأ</label>
   638→          <Input
   639→            dir="ltr"
   640→            className="num text-left h-10"
   641→            placeholder="اختیاری — مثال: 0101234567890"
   642→            value={accountNumber ? toFaDigits(accountNumber) : ""}
   643→            onChange={(e) => setAccountNumber(toEnDigits(e.target.value).replace(/[^\d]/g, "").slice(0, 26))}
   644→          />
   645→        </div>
   646→        <div className="space-y-1.5">
   647→          <label className="text-xs font-semibold">شماره کارت مبدأ</label>
   648→          <Input
   649→            dir="ltr"
   650→            inputMode="numeric"
   651→            className="num text-left h-10"
   652→            placeholder="اختیاری — ۱۶ رقم"
   653→            value={cardNumber ? formatCardNumber(cardNumber) : ""}
   654→            onChange={(e) => setCardNumber(digitsOnly(e.target.value).slice(0, 16))}
   655→          />
   656→        </div>
   657→        <div className="space-y-1.5 sm:col-span-2">
   658→          <label className="text-xs font-semibold flex items-center gap-1">
   659→            <Hash className="size-3.5 text-primary" />
   660→            شماره پیگیری/ارجاع انتقال
   661→          </label>
   662→          <Input
   663→            dir="ltr"
   664→            className="text-left h-10"
   665→            placeholder="اختیاری — شماره ارجاع بانک"
   666→            value={referenceNumber}
   667→            onChange={(e) => setReferenceNumber(e.target.value.slice(0, 40))}
   668→          />
   669→        </div>
   670→      </div>
   671→      <Button className="w-full h-12 font-bold" onClick={submit} disabled={submitting}>
   672→        {submitting ? (
   673→          <>
   674→            <Loader2 className="size-4 animate-spin" /> در حال ثبت…
   675→          </>
   676→        ) : (
   677→          <>
   678→            <CheckCircle2 className="size-4" /> ثبت پرداخت و بایگانی مرجوعی
   679→          </>
   680→        )}
   681→      </Button>
   682→    </section>
   683→  );
   684→}
