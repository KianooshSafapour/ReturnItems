   326→        target={resetTarget}
   327→        onOpenChange={(o) => {
   328→          if (!o) setResetTarget(null);
   329→        }}
   330→        onDone={onChanged}
   331→      />
   332→
   333→      {/* پنجره ویرایش کاربر */}
   334→      <EditUserDialog
   335→        target={editTarget}
   336→        user={user}
   337→        onOpenChange={(o) => {
   338→          if (!o) setEditTarget(null);
   339→        }}
   340→        onSaved={() => {
   341→          load();
   342→          onChanged?.();
   343→        }}
   344→      />
   345→    </div>
   346→  );
   347→}

   351→function AddUserDialog({
   352→  open,
   353→  onOpenChange,
   354→  onCreated,
   355→}: {
   356→  open: boolean;
   357→  onOpenChange: (o: boolean) => void;
   358→  onCreated: () => void;
   359→}) {
   360→  const [fullName, setFullName] = useState("");
   361→  const [username, setUsername] = useState("");
   362→  const [phone, setPhone] = useState("");
   363→  const [password, setPassword] = useState("");
   364→  const [role, setRole] = useState<Role>("cashier");
   365→  const [showPass, setShowPass] = useState(false);
   366→  const [errors, setErrors] = useState<FormErrors>({});
   367→  const [submitting, setSubmitting] = useState(false);
   368→
   369→  useEffect(() => {
   370→    if (open) {
   371→      setFullName("");
   372→      setUsername("");
   373→      setPhone("");
   374→      setPassword("");
   375→      setRole("cashier");
   376→      setShowPass(false);
   377→      setErrors({});
   378→    }
   379→  }, [open]);
   380→
   381→  const validate = (): boolean => {
   382→    const e: FormErrors = {};
   383→    if (fullName.trim().length < 3) e.fullName = "نام کامل حداقل ۳ نویسه باشد";
   384→    if (!/^[a-z0-9_.-]{3,30}$/.test(username.trim().toLowerCase()))
   385→      e.username = "نام کاربری باید ۳ تا ۳۰ نویسه لاتین کوچک، عدد، نقطه، خط تیره یا زیرخط باشد";
   386→    const ph = digitsOnly(toEnDigits(phone));
   387→    if (ph && !/^09\d{9}$/.test(ph)) e.phone = "شماره موبایل باید ۱۱ رقم و با ۰۹ شروع شود";
   388→    if (password.length < 6) e.password = "گذرواژه حداقل ۶ نویسه باشد";
   389→    setErrors(e);
   390→    return Object.keys(e).length === 0;
   391→  };
   392→
   393→  const submit = async () => {
   394→    if (!validate() || submitting) return;
   395→    setSubmitting(true);
   396→    try {
   397→      const res = await fetch("/api/users", {
   398→        method: "POST",
   399→        headers: { "Content-Type": "application/json" },
   400→        body: JSON.stringify({
   401→          fullName: fullName.trim(),
   402→          username: username.trim().toLowerCase(),
   403→          phone: digitsOnly(toEnDigits(phone)).trim(),
   404→          password,
   405→          role,
   406→        }),
   407→      });
   408→      const data = await res.json();
   409→      if (res.ok) {
   410→        toast.success(`حساب ${ROLE_LABELS[role]} «${fullName.trim()}» ایجاد شد`);
   411→        onOpenChange(false);
   412→        onCreated();
   413→      } else {
   414→        toast.error(data.error || "خطا در ایجاد حساب کاربری");
   415→      }
   416→    } catch {
   417→      toast.error("خطا در ایجاد حساب کاربری");
   418→    } finally {
   419→      setSubmitting(false);
   420→    }
   421→  };
   422→
   423→  return (
   424→    <Dialog open={open} onOpenChange={(o) => !submitting && onOpenChange(o)}>
   425→      <DialogContent dir="rtl" className="w-[calc(100vw-1.5rem)] sm:max-w-lg">
   426→        <DialogHeader>
   427→          <DialogTitle className="flex items-center gap-2">
   428→            <UserPlus className="size-5 text-primary" />
   429→            افزودن کاربر جدید
   430→          </DialogTitle>
   431→          <DialogDescription>
   432→            حساب تازه با نقش انتخابی ساخته می‌شود؛ نقش هر کاربر بعدها نیز قابل تغییر است.
   433→          </DialogDescription>
   434→        </DialogHeader>
   435→
   436→        <div className="space-y-4">
   437→          <div className="space-y-2">
   438→            <Label>نقش کاربر</Label>
   439→            <Select value={role} onValueChange={(v) => setRole(v as Role)}>
   440→              <SelectTrigger className="h-11 w-full" id="new-user-role" aria-label="نقش کاربر">
   441→                <SelectValue />
   442→              </SelectTrigger>
   443→              <SelectContent>
   444→                <SelectItem value="cashier">صندوقدار</SelectItem>
   445→                <SelectItem value="gm">مدیر کل (تأیید مرجوعی بدون پرداخت)</SelectItem>
   446→                <SelectItem value="owner">مدیر فروشگاه</SelectItem>
   447→                <SelectItem value="admin">مدیر ارشد سامانه</SelectItem>
   448→              </SelectContent>
   449→            </Select>
   450→            <p className="text-[11px] leading-5 text-muted-foreground">{ROLE_HINTS[role]}</p>
   451→          </div>
   452→
   453→          <div className="space-y-2">
   454→            <Label htmlFor="cashier-fullname">نام کامل</Label>
   455→            <Input
   456→              id="cashier-fullname"
   457→              placeholder="مثلاً: فاطمه رضایی"
   458→              value={fullName}
   459→              onChange={(e) => setFullName(e.target.value)}
   460→              className="h-11"
   461→            />
   462→            {errors.fullName && <p className="text-xs text-destructive">{errors.fullName}</p>}
   463→          </div>
   464→
   465→          <div className="space-y-2">
   466→            <Label htmlFor="cashier-username">نام کاربری</Label>
   467→            <Input
   468→              id="cashier-username"
   469→              placeholder="مثلاً: fateme.rezaei"
   470→              value={username}
   471→              onChange={(e) => setUsername(e.target.value)}
   472→              className="h-11"
   473→              dir="ltr"
   474→              autoComplete="off"
   475→            />
   476→            {errors.username ? (
   477→              <p className="text-xs text-destructive">{errors.username}</p>
   478→            ) : (
   479→              <p className="text-[11px] text-muted-foreground">
   480→                فقط حروف لاتین کوچک، عدد و نویسه‌های . _ -
   481→              </p>
   482→            )}
   483→          </div>

   543→function ResetPasswordDialog({
   544→  target,
   545→  onOpenChange,
   546→  onDone,
   547→}: {
   548→  target: UserRow | null;
   549→  onOpenChange: (o: boolean) => void;
   550→  onDone?: () => void;
   551→}) {
   552→  const [password, setPassword] = useState("");
   553→  const [showPass, setShowPass] = useState(false);
   554→  const [submitting, setSubmitting] = useState(false);
   555→
   556→  useEffect(() => {
   557→    setPassword("");
   558→    setShowPass(false);
   559→  }, [target]);
   560→
   561→  const submit = async () => {
   562→    if (!target || submitting) return;
   563→    if (password.length < 6) {
   564→      toast.error("گذرواژه حداقل ۶ نویسه باشد");
   565→      return;
   566→    }
   567→    setSubmitting(true);
   568→    try {
   569→      const res = await fetch(`/api/users/${target.id}`, {
   570→        method: "PATCH",
   571→        headers: { "Content-Type": "application/json" },
   572→        body: JSON.stringify({ password }),
   573→      });
   574→      const data = await res.json();
   575→      if (res.ok) {
   576→        toast.success(`گذرواژه «${target.fullName}» بازنشانی شد`);
   577→        onOpenChange(false);
   578→        onDone?.();
   579→      } else {
   580→        toast.error(data.error || "خطا در بازنشانی گذرواژه");
   581→      }
   582→    } catch {
   583→      toast.error("خطا در بازنشانی گذرواژه");
   584→    } finally {
   585→      setSubmitting(false);
   586→    }
   587→  };
   588→
   589→  return (
   590→    <Dialog open={!!target} onOpenChange={(o) => !submitting && onOpenChange(o)}>
   591→      <DialogContent dir="rtl" className="w-[calc(100vw-1.5rem)] sm:max-w-lg">
   592→        <DialogHeader>
   593→          <DialogTitle className="flex items-center gap-2">
   594→            <KeyRound className="size-5 text-primary" />
   595→            بازنشانی گذرواژه
   596→          </DialogTitle>
   597→          <DialogDescription>
   598→            گذرواژه جدید برای حساب «{target?.fullName}» (
   599→            <span className="num" dir="ltr">
   600→              {target?.username}
   601→            </span>
   602→            ) تعیین کنید.
   603→          </DialogDescription>
   604→        </DialogHeader>
   605→
   606→        <div className="space-y-2">
   607→          <Label htmlFor="reset-password">گذرواژه جدید</Label>
   608→          <div className="relative">
   609→            <Input
   610→              id="reset-password"
   611→              type={showPass ? "text" : "password"}
   612→              placeholder="حداقل ۶ نویسه"
   613→              value={password}
   614→              onChange={(e) => setPassword(e.target.value)}
   615→              onKeyDown={(e) => e.key === "Enter" && submit()}
   616→              className="h-11 pe-11"
   617→              dir="ltr"
   618→              autoComplete="new-password"
   619→            />
   620→            <button
   621→              type="button"
   622→              onClick={() => setShowPass((s) => !s)}
   623→              className="absolute end-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground transition-colors"
   624→              aria-label={showPass ? "پنهان‌کردن گذرواژه" : "نمایش گذرواژه"}
   625→            >
   626→              {showPass ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   627→            </button>
   628→          </div>
   629→          <p className="text-[11px] text-muted-foreground leading-5">
   630→            پس از بازنشانی، صندوقدار باید در ورود بعدی از گذرواژه جدید استفاده کند.
   631→          </p>
   632→        </div>
   633→
   634→        <DialogFooter className="gap-2">
   635→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   636→            انصراف
   637→          </Button>
   638→          <Button onClick={submit} disabled={submitting} className="gap-1.5">
   639→            {submitting && <Loader2 className="size-4 animate-spin" />}
   640→            بازنشانی گذرواژه
   641→          </Button>
   642→        </DialogFooter>
   643→      </DialogContent>
   644→    </Dialog>
   645→  );
   646→}

   650→function EditUserDialog({
   651→  target,
   652→  user,
   653→  onOpenChange,
   654→  onSaved,
   655→}: {
   656→  target: UserRow | null;
   657→  user: SessionUser;
   658→  onOpenChange: (o: boolean) => void;
   659→  onSaved: () => void;
   660→}) {
   661→  const [fullName, setFullName] = useState("");
   662→  const [username, setUsername] = useState("");
   663→  const [phone, setPhone] = useState("");
   664→  const [password, setPassword] = useState("");
   665→  const [role, setRole] = useState<Role>("cashier");
   666→  const [showPass, setShowPass] = useState(false);
   667→  const [isActive, setIsActive] = useState(true);
   668→  const [errors, setErrors] = useState<FormErrors>({});
   669→  const [submitting, setSubmitting] = useState(false);
   670→
   671→  useEffect(() => {
   672→    if (target) {
   673→      setFullName(target.fullName);
   674→      setUsername(target.username);
   675→      setPhone(target.phone ?? "");
   676→      setPassword("");
   677→      setRole(target.role);
   678→      setShowPass(false);
   679→      setIsActive(target.isActive);
   680→      setErrors({});
   681→    }
   682→  }, [target]);
   683→
   684→  const validate = (): boolean => {
   685→    const e: FormErrors = {};
   686→    if (fullName.trim().length < 3) e.fullName = "نام کامل حداقل ۳ نویسه باشد";
   687→    if (!/^[a-z0-9_.-]{3,30}$/.test(username.trim().toLowerCase()))
   688→      e.username = "نام کاربری باید ۳ تا ۳۰ نویسه لاتین کوچک، عدد، نقطه، خط تیره یا زیرخط باشد";
   689→    const ph = digitsOnly(toEnDigits(phone));
   690→    if (ph && !/^09\d{9}$/.test(ph)) e.phone = "شماره موبایل باید ۱۱ رقم و با ۰۹ شروع شود";
   691→    if (password && password.length < 6) e.password = "گذرواژه جدید حداقل ۶ نویسه باشد";
   692→    setErrors(e);
   693→    return Object.keys(e).length === 0;
   694→  };
   695→
   696→  const submit = async () => {
   697→    if (!target || submitting) return;
   698→    if (!validate()) return;
   699→
   700→    // تنها فیلدهای تغییر‌یافته ارسال می‌شود
   701→    const fullNameT = fullName.trim();
   702→    const usernameT = username.trim().toLowerCase();
   703→    const phoneT = digitsOnly(toEnDigits(phone));
   704→    const payload: Record<string, string | boolean> = {};
   705→    if (fullNameT !== target.fullName) payload.fullName = fullNameT;
   706→    if (usernameT !== target.username) payload.username = usernameT;
   707→    if (phoneT !== (target.phone ?? "")) payload.phone = phoneT; // رشته خالی = پاک‌کردن
   708→    if (password) payload.password = password;
   709→    if (role !== target.role) payload.role = role;
   710→    if (!isManagerRole(target.role) && target.id !== user.id && isActive !== target.isActive)
   711→      payload.isActive = isActive;
   712→
   713→    if (Object.keys(payload).length === 0) {
   714→      toast.info("تغییری برای ذخیره وجود ندارد");
   715→      return;
   716→    }
   717→
   718→    const roleChanged = payload.role !== undefined;
   719→
   720→    setSubmitting(true);
   721→    try {
   722→      const res = await fetch(`/api/users/${target.id}`, {
   723→        method: "PATCH",
   724→        headers: { "Content-Type": "application/json" },
   725→        body: JSON.stringify(payload),
   726→      });
   727→      const data = await res.json();
   728→      if (res.ok) {
   729→        toast.success(
   730→          roleChanged
   731→            ? `کاربر به‌روزرسانی شد — نقش جدید: ${ROLE_LABELS[role]}`
   732→            : "کاربر به‌روزرسانی شد"
   733→        );
   734→        onOpenChange(false);
   735→        onSaved();
   736→      } else {
   737→        toast.error(data.error || "خطا در به‌روزرسانی کاربر");
   738→      }
   739→    } catch {
   740→      toast.error("خطا در به‌روزرسانی کاربر");
   741→    } finally {
   742→      setSubmitting(false);
   743→    }
   744→  };
   745→
   746→  const isManagerRow = !!target && isManagerRole(target.role);
   747→  const isOwnRow = !!target && target.id === user.id;
   748→
   749→  return (
   750→    <Dialog open={!!target} onOpenChange={(o) => !submitting && onOpenChange(o)}>
   751→      <DialogContent
   752→        dir="rtl"
   753→        className="w-[calc(100vw-1.5rem)] sm:max-w-lg max-h-[calc(100dvh-2rem)] overflow-y-auto"
   754→      >
   755→        <DialogHeader>
   756→          <DialogTitle className="flex items-center gap-2">
   757→            <Pencil className="size-5 text-primary" />
   758→            ویرایش کاربر
   759→          </DialogTitle>
   760→          <DialogDescription>
   761→            ویرایش مشخصات و نقش حساب «{target?.fullName}» — تنها فیلدهای تغییر‌یافته ذخیره می‌شوند.
   762→          </DialogDescription>
   763→        </DialogHeader>
   764→
   765→        <div className="space-y-4">
   766→          <div className="space-y-2">
   767→            <Label>نقش کاربر</Label>
   768→            <Select
   769→              value={role}
   770→              onValueChange={(v) => setRole(v as Role)}
   771→              disabled={isOwnRow}
   772→            >
   773→              <SelectTrigger className="h-11 w-full" id="edit-user-role" aria-label="نقش کاربر">

   786→                نقش حساب خودتان قابل تغییر نیست.
