    88→  function addItem(p: ProductDTO | null, manual = false): boolean {
    89→    let existing: ReturnItemDraft | undefined;
    90→    if (p) {
    91→      existing = items.find(
    92→        (i) => (p.barcode && i.barcode === p.barcode) || (p.id && i.productId === p.id)
    93→      );
    94→      if (existing) {
    95→        setItems((prev) =>
    96→          prev.map((i) => (i.key === existing!.key ? { ...i, quantity: Math.min(999, i.quantity + 1) } : i))
    97→        );
    98→        toast.info(`«${p.name}» از قبل در فهرست است — تعداد افزایش یافت`, {
    99→          description: "تعداد فعلی: " + toFaDigits((existing.quantity || 0) + 1),
   100→        });
   101→        return true;
   102→      }
   103→      setItems((prev) => [
   104→        {
   105→          key: `k${keySeq++}`,
   106→          productId: p.id,
   107→          name: p.name,
   108→          barcode: p.barcode,
   109→          code: p.code,
   110→          quantity: 1,
   111→          unitPrice: p.price,
   112→          fromDb: true,
   113→        },
   114→        ...prev,
   115→      ]);
   116→      return true;
   117→    }
   118→    if (manual) {
   119→      setItems((prev) => [
   120→        {
   121→          key: `k${keySeq++}`,
   122→          productId: null,
   123→          name: "",
   124→          barcode: null,
   125→          code: null,
   126→          quantity: 1,
   127→          unitPrice: 0,
   128→          fromDb: false,
   129→        },
   130→        ...prev,
   131→      ]);
   132→      return true;
   133→    }
   134→    return false;
   135→  }

   137→  async function handleBarcode(e: React.KeyboardEvent<HTMLInputElement>) {
   138→    if (e.key !== "Enter") return;
   139→    e.preventDefault();
   140→    const code = digitsOnly(barcodeInput);
   141→    if (code.length < 4) {
   142→      if (code.length > 0) toast.error("بارکد خیلی کوتاه است");
   143→      return;
   144→    }
   145→    setScanning(true);
   146→    try {
   147→      const res = await fetch(`/api/products?barcode=${encodeURIComponent(code)}`);
   148→      const data = await res.json();
   149→      const p: ProductDTO | undefined = data.products?.[0];
   150→      if (p) {
   151→        addItem(p);
   152→        toast.success(`افزوده شد: ${p.name}`, {
   153→          description: `قیمت: ${formatMoney(p.price)} ریال`,
   154→        });
   155→        setBarcodeInput("");
   156→      } else {
   157→        toast.error(`کالایی با بارکد ${toFaDigits(code)} در بانک کالا یافت نشد`, {
   158→          description: "می‌توانید آن را به‌صورت دستی به فهرست اضافه کنید",
   159→          duration: 5000,
   160→        });
   161→      }
   162→    } catch {
   163→      toast.error("خطا در جستجوی بارکد");
   164→    } finally {
   165→      setScanning(false);
   166→      barcodeRef.current?.focus();
   167→    }
   168→  }

   170→  function updateItem(key: string, patch: Partial<ReturnItemDraft>) {
   171→    setItems((prev) => prev.map((i) => (i.key === key ? { ...i, ...patch } : i)));
   172→  }
   173→  function removeItem(key: string) {
   174→    setItems((prev) => prev.filter((i) => i.key !== key));
   175→  }

   177→  function validate(): string | null {
   178→    if (customerName.trim().length < 2) return "نام مشتری را کامل وارد کنید";
   179→    if (!phoneOk) return "شماره موبایل مشتری معتبر نیست (۰۹xxxxxxxxx)";
   180→    if (!cardOk) return "شماره کارت بانکی مشتری معتبر نیست";
   181→    if (!purchaseAt) return "زمان خرید مشتری را انتخاب کنید";
   182→    if (!reason) return "دلیل مرجوعی را انتخاب کنید";
   183→    if (items.length === 0) return "حداقل یک کالا به فهرست اضافه کنید";
   184→    for (const i of items) {
   185→      if (!i.name.trim()) return "نام یکی از اقلام خالی است";
   186→      if (i.quantity < 1) return `تعداد «${i.name}» باید حداقل ۱ باشد`;
   187→      if (i.unitPrice <= 0) return `قیمت «${i.name}» را وارد کنید`;
   188→    }
   189→    return null;
   190→  }

   192→  async function handleSubmit(e?: React.FormEvent) {
   193→    e?.preventDefault();
   194→    if (submitting) return;
   195→    const err = validate();
   196→    if (err) {
   197→      toast.error(err, { description: "لطفاً فرم را کامل کنید" });
   198→      return;
   199→    }
   200→    setSubmitting(true);
   201→    try {
   202→      const res = await fetch("/api/returns", {
   203→        method: "POST",
   204→        headers: { "Content-Type": "application/json" },
   205→        body: JSON.stringify({
   206→          customerName: customerName.trim(),
   207→          customerPhone: digitsOnly(customerPhone),
   208→          customerCardNumber: digitsOnly(customerCard),
   209→          purchaseAt: purchaseAt?.toISOString(),
   210→          reason: reason === "سایر" ? note.trim() || "سایر" : reason,
   211→          note: note.trim() || null,
   212→          items: items.map((i) => ({
   213→            productId: i.productId,
   214→            name: i.name.trim(),
   215→            barcode: i.barcode,
   216→            code: i.code,
   217→            quantity: i.quantity,
   218→            unitPrice: i.unitPrice,
   219→          })),
   220→        }),
   221→      });
   222→      const data = await res.json();
   223→      if (!res.ok) {
   224→        toast.error(data.error || "ثبت مرجوعی ناموفق بود");
   225→        return;
   226→      }
   227→      setSuccess({ code: data.code, total: data.totalAmount, count: data.itemCount, at: new Date(data.submittedAt) });
   228→      setTodayCount((c) => c + 1);
   229→      toast.success(`مرجوعی ${data.code} با موفقیت ثبت شد 🌿`, {
   230→        description: "به آقای نوروزی اطلاع داده شد تا وجه را انتقال دهد",
   231→      });
   232→    } catch {
   233→      toast.error("ارتباط با سرور برقرار نشد");
   234→    } finally {
   235→      setSubmitting(false);
   236→    }
   237→  }
