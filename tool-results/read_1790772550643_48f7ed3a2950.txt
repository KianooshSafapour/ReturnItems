    48→        status: status === "all" ? "all" : status,
    49→        page: String(page),
    50→        pageSize: String(pageSize),
    51→      });
    52→      if (q.trim()) params.set("q", q.trim());

    56→        setRows(data.returns ?? []);
    57→        setTotal(data.total ?? 0);

    60→      toast.error("خطا در دریافت فهرست مرجوعی‌ها");
    61→    } finally {
    62→      setLoading(false);
    63→    }

    66→  useEffect(() => {

    68→    return () => clearTimeout(t);

   120→                <tr className="text-muted-foreground text-xs">
   121→                  <th className="text-start font-semibold px-4 py-3">شماره پیگیری</th>
   122→                  <th className="text-start font-semibold px-4 py-3">مشتری</th>
   123→                  <th className="text-start font-semibold px-4 py-3">مبلغ کل</th>
   124→                  <th className="text-start font-semibold px-4 py-3">اقلام</th>

   127→                  )}
   128→                  <th className="text-start font-semibold px-4 py-3">زمان ثبت</th>
   129→                  <th className="text-start font-semibold px-4 py-3">وضعیت</th>

   212→        {/* صفحه‌بندی */}

   216→              نمایش {toFaDigits((page - 1) * pageSize + 1)} تا{" "}

   218→            </p>
   219→            <div className="flex items-center gap-1.5">
   220→              <Button
   221→                variant="outline"
   222→                size="icon"
   223→                className="size-8"
   224→                disabled={page <= 1}
   225→                onClick={() => setPage((p) => p - 1)}
   226→                aria-label="صفحه قبل"
   227→              >
   228→                <ChevronRight className="size-4" />
   229→              </Button>
   230→              <span className="num text-sm font-bold px-2">
   231→                {toFaDigits(page)} / {toFaDigits(pages)}
   232→              </span>
   233→              <Button
   234→                variant="outline"
   235→                size="icon"
   236→                className="size-8"
   237→                disabled={page >= pages}
   238→                onClick={() => setPage((p) => p + 1)}
   239→                aria-label="صفحه بعد"

   255→          onChanged?.();
   256→        }}
   257→      />

   262→// ---------- پنجره جزئیات مرجوعی ----------
   263→export function ReturnDetailDialog({
   264→  id,
   265→  onClose,
   266→  user,
   267→  onPaid,
   268→}: {
   269→  id: string | null;
   270→  onClose: () => void;
   271→  user: SessionUser;
   272→  onPaid?: () => void;
   273→}) {

   275→  const [loading, setLoading] = useState(false);
   276→
   277→  useEffect(() => {

   309→                  <Badge className="bg-chart-2/15 text-chart-2 border-chart-2/30 gap-1">
   310→                    <Hourglass className="size-3" /> در انتظار پرداخت
   311→                  </Badge>
   312→                ) : (
   313→                  <Badge className="bg-primary/15 text-primary border-primary/30 gap-1">
   314→                    <CheckCircle2 className="size-3" /> پرداخت و بایگانی شد
   315→                  </Badge>
   316→                )}
   317→              </DialogTitle>
   318→            </DialogHeader>
   319→
   320→            <div className="p-6 space-y-6">

   496→  icon: React.ReactNode;
   497→  label: string;
   498→  value: string;

   500→}) {
   501→  return (
   502→    <div className="flex items-center gap-3 rounded-lg bg-muted/40 px-3 py-2.5 min-w-0">
   503→      <span className="text-primary shrink-0">{icon}</span>

   521→export function PaymentForm({
   522→  detail,
   523→  onPaid,
   524→}: {
   525→  detail: ReturnDetailDTO;
   526→  onPaid: (code: string) => void;
   527→}) {
   528→  const [bankName, setBankName] = useState("");
   529→  const [accountNumber, setAccountNumber] = useState("");
   530→  const [cardNumber, setCardNumber] = useState("");
   531→  const [referenceNumber, setReferenceNumber] = useState("");
   532→  const [transferredAt, setTransferredAt] = useState<Date | null>(null);
   533→  const [submitting, setSubmitting] = useState(false);
   534→
   535→  async function submit() {
   536→    if (submitting) return;
   537→    if (!bankName) {
   538→      toast.error("بانک مبدأ را انتخاب کنید");
   539→      return;
   540→    }
   541→    if (!accountNumber && !digitsOnly(cardNumber) && !referenceNumber) {
   542→      toast.error("حداقل شماره حساب، کارت یا شماره پیگیری را وارد کنید");
   543→      return;
   544→    }
   545→    if (cardNumber && !/^\d{16}$/.test(digitsOnly(cardNumber))) {
   546→      toast.error("شماره کارت مبدأ باید ۱۶ رقم باشد");
   547→      return;
   548→    }
   549→    if (!transferredAt) {
   550→      toast.error("زمان انتقال وجه را انتخاب کنید");
   551→      return;
   552→    }
   553→    setSubmitting(true);
   554→    try {
   555→      const res = await fetch(`/api/returns/${detail.id}/payment`, {
   556→        method: "POST",
   557→        headers: { "Content-Type": "application/json" },
   558→        body: JSON.stringify({
   559→          bankName,
   560→          accountNumber: accountNumber.trim() || null,
   561→          cardNumber: digitsOnly(cardNumber) || null,
   562→          referenceNumber: referenceNumber.trim() || null,
   563→          transferredAt: transferredAt.toISOString(),
   564→        }),
   565→      });
   566→      const data = await res.json();
   567→      if (!res.ok) {
   568→        toast.error(data.error || "ثبت پرداخت ناموفق بود");
   569→        return;
   570→      }
   571→      onPaid(data.code);
   572→    } catch {
   573→      toast.error("ارتباط با سرور برقرار نشد");
   574→    } finally {
   575→      setSubmitting(false);
   576→    }
   577→  }
   578→
   579→  return (
   580→    <section className="rounded-xl border-2 border-primary/30 bg-card p-4 space-y-4">
   581→      <div className="flex items-center gap-2">
   582→        <Wallet className="size-4 text-primary" />
   583→        <h4 className="font-bold text-sm">ثبت بازگشت وجه (ویژه مدیر)</h4>
   584→      </div>
   585→      <IlluminatedDivider />
   586→      <div className="rounded-lg bg-primary/5 border border-primary/20 p-3">
   587→        <MoneyDisplay amount={detail.totalAmount} size="md" />
   588→      </div>
   589→      <div className="grid sm:grid-cols-2 gap-3">
   590→        <div className="space-y-1.5">
   591→          <label className="text-xs font-semibold flex items-center gap-1">
   592→            <Landmark className="size-3.5 text-primary" />
   593→            بانک مبدأ (انتقال وجه)
   594→          </label>
   595→          <Select value={bankName} onValueChange={setBankName}>
   596→            <SelectTrigger className="h-10">
   597→              <SelectValue placeholder="انتخاب بانک…" />
   598→            </SelectTrigger>
   599→            <SelectContent className="max-h-72">
   600→              {BANKS.map((b) => (
   601→                <SelectItem key={b} value={b}>
   602→                  {b}
   603→                </SelectItem>
   604→              ))}
   605→            </SelectContent>
   606→          </Select>
   607→        </div>
   608→        <div className="space-y-1.5">
   609→          <label className="text-xs font-semibold">زمان انتقال وجه</label>
   610→          <JalaliDateTimePicker value={transferredAt} onChange={setTransferredAt} placeholder="انتخاب زمان انتقال" />
   611→        </div>
   612→        <div className="space-y-1.5">
   613→          <label className="text-xs font-semibold">شماره حساب مبدأ</label>
   614→          <Input
   615→            dir="ltr"
   616→            className="num text-left h-10"
   617→            placeholder="اختیاری — مثال: 0101234567890"
   618→            value={accountNumber ? toFaDigits(accountNumber) : ""}
   619→            onChange={(e) => setAccountNumber(toEnDigits(e.target.value).replace(/[^\d]/g, "").slice(0, 26))}
   620→          />
   621→        </div>
   622→        <div className="space-y-1.5">
   623→          <label className="text-xs font-semibold">شماره کارت مبدأ</label>
   624→          <Input
   625→            dir="ltr"
   626→            inputMode="numeric"
   627→            className="num text-left h-10"
   628→            placeholder="اختیاری — ۱۶ رقم"
   629→            value={cardNumber ? formatCardNumber(cardNumber) : ""}
   630→            onChange={(e) => setCardNumber(digitsOnly(e.target.value).slice(0, 16))}
   631→          />
   632→        </div>
   633→        <div className="space-y-1.5 sm:col-span-2">
   634→          <label className="text-xs font-semibold flex items-center gap-1">
   635→            <Hash className="size-3.5 text-primary" />
   636→            شماره پیگیری/ارجاع انتقال
   637→          </label>
   638→          <Input
   639→            dir="ltr"
   640→            className="text-left h-10"
   641→            placeholder="اختیاری — شماره ارجاع بانک"
   642→            value={referenceNumber}
   643→            onChange={(e) => setReferenceNumber(e.target.value.slice(0, 40))}
   644→          />
   645→        </div>
   646→      </div>
   647→      <Button className="w-full h-12 font-bold" onClick={submit} disabled={submitting}>
   648→        {submitting ? (
   649→          <>
   650→            <Loader2 className="size-4 animate-spin" /> در حال ثبت…
   651→          </>
   652→        ) : (
   653→          <>
   654→            <CheckCircle2 className="size-4" /> ثبت پرداخت و بایگانی مرجوعی
   655→          </>
   656→        )}
   657→      </Button>
   658→    </section>
   659→  );
   660→}

   666→    <Button
   667→      variant="outline"
   668→      className="h-11 font-semibold"

   675→            const d = await res.json().catch(() => ({}));

   682→          a.href = url;
   683→          a.download = "hyper-zeytoon-returns.xlsx";
   684→          a.click();
   685→          URL.revokeObjectURL(url);
   686→          toast.success("فایل اکسل آرشیو مرجوعی‌ها دانلود شد");
   687→        } catch {
   688→          toast.error("دانلود خروجی ناموفق بود");
   689→        } finally {
   690→          setLoading(false);
   691→        }
   692→      }}
   693→    >
   694→      {loading ? <Loader2 className="size-4 animate-spin" /> : <Download className="size-4" />}
