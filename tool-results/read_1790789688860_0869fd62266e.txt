   168→  function addItem(p: ProductDTO | null, manual = false): boolean {
   169→    let existing: ReturnItemDraft | undefined;

   171→      existing = items.find(
   172→        (i) => (p.barcode && i.barcode === p.barcode) || (p.id && i.productId === p.id)
   173→      );
   174→      if (existing) {
   175→        setItems((prev) =>

   181→        return true;
   182→      }
   183→      setItems((prev) => [
   184→        {
   185→          key: `k${keySeq++}`,
   186→          productId: p.id,
   187→          name: p.name,
   188→          barcode: p.barcode,
   189→          code: p.code,

   191→          unitPrice: p.price,
   192→          fromDb: true,
   193→          barcodeLocked: !!p.barcode,
   194→        },
   195→        ...prev,
   196→      ]);
   197→      return true;
   198→    }

   202→          key: `k${keySeq++}`,

   275→  function validate(): string | null {
   276→    if (customerName.trim().length < 2) return "نام مشتری را کامل وارد کنید";
   277→    if (!phoneOk) return "شماره موبایل مشتری معتبر نیست (۰۹xxxxxxxxx)";
   278→    if (!cardOk) return "شماره کارت بانکی مشتری معتبر نیست";
   279→    if (!purchaseAt) return "زمان خرید مشتری را انتخاب کنید";
   280→    if (!reason) return "دلیل مرجوعی را انتخاب کنید";
   281→    if (items.length === 0) return "حداقل یک کالا به فهرست اضافه کنید";
   282→    for (const i of items) {
   283→      if (!i.name.trim()) return "نام یکی از اقلام خالی است";
   284→      if (i.quantity < 1) return `تعداد «${i.name}» باید حداقل ۱ باشد`;
   285→      if (i.unitPrice <= 0) return `قیمت «${i.name}» را وارد کنید`;
   286→      if (!i.barcode || digitsOnly(i.barcode).length < 4)
   287→        return `بارکد «${i.name}» خالی است — بارکد را وارد کنید یا دکمه تولید بارکد داخلی را بزنید`;
   288→    }
   289→    // منطق تسویه
   290→    if (exchangeValue > total && transferAmount > 0)
   291→      return "در حالت بدهی مشتری (جایگزین بیشتر از مرجوعی)، مبلغ قابل انتقال باید صفر باشد";
   292→    if (exchangeValue <= total && transferAmount > total - exchangeValue)
   293→      return "مبلغ قابل انتقال نمی‌تواند از «جمع اقلام منهای ارزش جایگزین» بیشتر باشد";
   294→    if (noteRequired && transferNote.trim().length < 3)
   295→      return "دلیل اختلاف مبلغ انتقالی با جمع اقلام را در توضیح تسویه بنویسید";
   296→    return null;
   297→  }

   299→  async function handleSubmit(e?: React.FormEvent) {
   300→    e?.preventDefault();
   301→    if (submitting) return;
   302→    const err = validate();
   303→    if (err) {
   304→      toast.error(err, { description: "لطفاً فرم را کامل کنید" });
   305→      return;
   306→    }
   307→    setSubmitting(true);
   308→    try {
   309→      const res = await fetch("/api/returns", {
   310→        method: "POST",
   311→        headers: { "Content-Type": "application/json" },
   312→        body: JSON.stringify({
   313→          customerName: customerName.trim(),
   314→          customerPhone: digitsOnly(customerPhone),
   315→          customerCardNumber: digitsOnly(customerCard),
   316→          purchaseAt: purchaseAt?.toISOString(),
   317→          reason: reason === "سایر" ? note.trim() || "سایر" : reason,
   318→          note: note.trim() || null,
   319→          exchangeValue,
   320→          transferAmount,
   321→          transferNote: transferNote.trim() || null,
   322→          items: items.map((i) => ({
   323→            productId: i.productId,
   324→            name: i.name.trim(),
   325→            barcode: digitsOnly(i.barcode ?? "") || null,
   326→            code: i.code,
   327→            quantity: i.quantity,
   328→            unitPrice: i.unitPrice,
   329→          })),
   330→        }),
   331→      });
   332→      const data = await res.json();
   333→      if (!res.ok) {
   334→        toast.error(data.error || "ثبت مرجوعی ناموفق بود");
   335→        return;
   336→      }
   337→      setSuccess({
   338→        id: data.id,
   339→        code: data.code,
   340→        total: data.totalAmount,
   341→        transfer: data.transferAmount ?? data.totalAmount,
   342→        exchange: data.exchangeValue ?? 0,
   343→        settlementType: (data.settlementType ?? "full") as SettlementType,
   344→        count: data.itemCount,
   345→        at: new Date(data.submittedAt),
   346→      });
   347→      setTodayCount((c) => c + 1);
   348→      toast.success(`مرجوعی ${data.code} با موفقیت ثبت شد 🌿`, {
   349→        description: "به آقای نوروزی اطلاع داده شد تا وجه را انتقال دهد",
   350→      });
   351→    } catch {
   352→      toast.error("ارتباط با سرور برقرار نشد");
   353→    } finally {
   354→      setSubmitting(false);
   355→    }
   356→  }
