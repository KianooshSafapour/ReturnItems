   316→   316→
   317→   317→function AddCashierDialog({
   318→   318→  open,
   319→   319→  onOpenChange,
   320→   320→  onCreated,
   321→   321→}: {
   322→   322→  open: boolean;
   323→   323→  onOpenChange: (o: boolean) => void;
   324→   324→  onCreated: () => void;
   325→   325→}) {
   326→   326→  const [fullName, setFullName] = useState("");
   327→   327→  const [username, setUsername] = useState("");
   328→   328→  const [phone, setPhone] = useState("");
   329→   329→  const [password, setPassword] = useState("");
   330→   330→  const [showPass, setShowPass] = useState(false);
   331→   331→  const [errors, setErrors] = useState<FormErrors>({});
   332→   332→  const [submitting, setSubmitting] = useState(false);
   333→   333→
   334→   334→  useEffect(() => {
   335→   335→    if (open) {
   336→   336→      setFullName("");
   337→   337→      setUsername("");
   338→   338→      setPhone("");
   339→   339→      setPassword("");
   340→   340→      setShowPass(false);
   341→   341→      setErrors({});
   342→   342→    }
   343→   343→  }, [open]);
   344→   344→
   345→   345→  const validate = (): boolean => {
   346→   346→    const e: FormErrors = {};
   347→   347→    if (fullName.trim().length < 3) e.fullName = "نام کامل حداقل ۳ نویسه باشد";
   348→   348→    if (!/^[a-z0-9_.-]{3,30}$/.test(username.trim().toLowerCase()))
   349→   349→      e.username = "نام کاربری باید ۳ تا ۳۰ نویسه لاتین کوچک، عدد، نقطه، خط تیره یا زیرخط باشد";
   350→   350→    const ph = digitsOnly(toEnDigits(phone));
   351→   351→    if (ph && !/^09\d{9}$/.test(ph)) e.phone = "شماره موبایل باید ۱۱ رقم و با ۰۹ شروع شود";
   352→   352→    if (password.length < 6) e.password = "گذرواژه حداقل ۶ نویسه باشد";
   353→   353→    setErrors(e);
   354→   354→    return Object.keys(e).length === 0;
   355→   355→  };
   356→   356→
   357→   357→  const submit = async () => {
   358→   358→    if (!validate() || submitting) return;
   359→   359→    setSubmitting(true);
   360→   360→    try {
   361→   361→      const res = await fetch("/api/users", {
   362→   362→        method: "POST",
   363→   363→        headers: { "Content-Type": "application/json" },
   364→   364→        body: JSON.stringify({
   365→   365→          fullName: fullName.trim(),
   366→   366→          username: username.trim().toLowerCase(),
   367→   367→          phone: digitsOnly(toEnDigits(phone)).trim(),
   368→   368→          password,
   369→   369→        }),
   370→   370→      });
   371→   371→      const data = await res.json();
   372→   372→      if (res.ok) {
   373→   373→        toast.success(`حساب صندوقدار «${fullName.trim()}» ایجاد شد`);
   374→   374→        onOpenChange(false);
   375→   375→        onCreated();
   376→   376→      } else {
   377→   377→        toast.error(data.error || "خطا در ایجاد حساب کاربری");
   378→   378→      }
   379→   379→    } catch {
   380→   380→      toast.error("خطا در ایجاد حساب کاربری");
   381→   381→    } finally {
   382→   382→      setSubmitting(false);
   383→   383→    }
   384→   384→  };
   385→   385→
   386→   386→  return (
   387→   387→    <Dialog open={open} onOpenChange={(o) => !submitting && onOpenChange(o)}>
   388→   388→      <DialogContent dir="rtl" className="sm:max-w-md">
   389→   389→        <DialogHeader>
   390→   390→          <DialogTitle className="flex items-center gap-2">
   391→   391→            <UserPlus className="size-5 text-primary" />
   392→   392→            افزودن صندوقدار جدید
   393→   393→          </DialogTitle>
   394→   394→          <DialogDescription>
   395→   395→            حساب تازه با نقش «صندوقدار» ساخته می‌شود؛ ثبت مرجوعی بدون دسترسی به پرداخت و مدیریت.
   396→   396→          </DialogDescription>
   397→   397→        </DialogHeader>
   398→   398→
   399→   399→        <div className="space-y-4">
   400→   400→          <div className="space-y-2">
   401→   401→            <Label htmlFor="cashier-fullname">نام کامل</Label>
   402→   402→            <Input
   403→   403→              id="cashier-fullname"
   404→   404→              placeholder="مثلاً: فاطمه رضایی"
   405→   405→              value={fullName}
   406→   406→              onChange={(e) => setFullName(e.target.value)}
   407→   407→              className="h-11"
   408→   408→            />
   409→   409→            {errors.fullName && <p className="text-xs text-destructive">{errors.fullName}</p>}
   410→   410→          </div>
   411→   411→
   412→   412→          <div className="space-y-2">
   413→   413→            <Label htmlFor="cashier-username">نام کاربری</Label>
   414→   414→            <Input
   415→   415→              id="cashier-username"
   416→   416→              placeholder="مثلاً: fateme.rezaei"
   417→   417→              value={username}
   418→   418→              onChange={(e) => setUsername(e.target.value)}
   419→   419→              className="h-11"
   420→   420→              dir="ltr"
   421→   421→              autoComplete="off"
   422→   422→            />
   423→   423→            {errors.username ? (
   424→   424→              <p className="text-xs text-destructive">{errors.username}</p>
